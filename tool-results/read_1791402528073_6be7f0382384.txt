   365→class _BalancesCard extends StatelessWidget {

   368→  final PartyDetailState state;
   369→
   370→  @override
   371→  Widget build(BuildContext context) {
   372→    final l10n = AppLocalizations.of(context)!;
   373→    final scheme = Theme.of(context).colorScheme;
   374→    if (state.balances.isEmpty) {
   375→      return FinCard(
   376→        child: Text(
   377→          l10n.partyDetailBalancesEmpty,
   378→          style: Theme.of(context).textTheme.bodySmall

   381→      );
   382→    }
   383→    return FinCard(
   384→      padding: const EdgeInsets.symmetric(horizontal: 6, vertical: 6),
   385→      child: Column(
   386→        children: [
   387→          for (var i = 0; i < state.balances.length; i++) ...[
   388→            if (i > 0)
   389→              Divider(
   390→                height: 1,
   391→                color: scheme.outlineVariant.withValues(alpha: 0.4),
   392→              ),
   393→            _BalanceRow(
   394→              row: state.balances[i],
   395→              selected:
   396→                  _currencyIdOfCode(state, state.balances[i].currencyCode) ==
   397→                  state.statementCurrencyId,
   398→              decimals: state.decimalsFor(state.balances[i].currencyCode),
   399→              onTap: () => _selectCurrency(
   400→                context,
   401→                _currencyIdOfCode(state, state.balances[i].currencyCode),
   402→              ),
   403→            ),
   404→          ],
   405→        ],
   406→      ),
   407→    );
   408→  }
   409→
   410→  static int? _currencyIdOfCode(PartyDetailState state, String code) {
   411→    for (final currency in state.currencies) {
   412→      if (currency.code == code) return currency.id;
   413→    }
   414→    return null;
   415→  }
   416→
   417→  static void _selectCurrency(BuildContext context, int? currencyId) {
   418→    final vm = context.read<PartyDetailViewModel>();
   419→    if (currencyId != null) {
   420→      unawaited(vm.setStatementCurrency(currencyId));
   421→    }
   422→  }
   423→}

   758→        l10n.partyDetailNoEntries,

   776→          code: statement.currencyCode,
   777→          amount: statement.finalBalance,
   778→          decimals: decimals,
   779→        ),
   780→        const SizedBox(height: 14),
   781→        if (statement.entries.isEmpty)
   782→          Text(
   783→            l10n.partyDetailNoEntries,
   784→            style: Theme.of(context).textTheme.bodySmall,

   789→              padding: const EdgeInsets.only(bottom: 10),
