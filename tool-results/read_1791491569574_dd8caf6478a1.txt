  1002→/// حجم القاعدة مقروءاً (ك.ب/م.ب) بأرقام النظام الحي — لبند «حول».
  1003→String _formatDbSize(BuildContext context, int bytes) {
  1004→  final l10n = AppLocalizations.of(context)!;
  1005→  final arabicIndic = NumeralsScope.of(context);
  1006→  String digits(String text) =>
  1007→      arabicIndic ? Numerals.toArabicIndic(text) : text;
  1008→  final kb = bytes / 1024;
  1009→  if (kb < 1024) {
  1010→    return l10n.backupSizeKb(digits(kb.round().toString()));
  1011→  }
  1012→  return l10n.backupSizeMb(digits((kb / 1024).toStringAsFixed(1)));
  1013→}
