   291→/// محرر رقم عشري بنافذة سفلية — للسعر/الكمية/الخصم/المبلغ النقدي.
   292→Future<double?> showNumberEditSheet(
   293→  BuildContext context, {
   294→  required String title,
   295→  required double initial,
   296→  required String confirmLabel,
   297→  String? hint,
   298→  bool allowZero = true,
   299→  int decimals = 2,
   300→  IconData? icon,
   301→}) {
   302→  final controller = TextEditingController(
   303→    text: initial == initial.truncateToDouble()
   304→        ? initial.truncate().toString()
   305→        : initial.toStringAsFixed(decimals),
   306→  );
   307→  String? error;
   308→  return showModalBottomSheet<double>(
   309→    context: context,
   310→    isScrollControlled: true,
   311→    // فوق شريط التبويبات: تُفتح على متصفح الفرع لا الجذر — تنزلق
   312→    // من فوق الشريط السفلي بدل أن تغطيه من أسفل الشاشة.
   313→    useRootNavigator: false,
   314→    builder: (sheetContext) {
   315→      final l10n = AppLocalizations.of(sheetContext)!;
   316→      final scheme = Theme.of(sheetContext).colorScheme;
   317→      return StatefulBuilder(
   318→        builder: (sheetContext, setSheetState) {
   319→          return Padding(
   320→            padding: EdgeInsets.only(
   321→              bottom: MediaQuery.of(sheetContext).viewInsets.bottom,
   322→            ),
   323→            child: Padding(
   324→              padding: const EdgeInsets.fromLTRB(20, 20, 20, 24),
   325→              child: Column(
   326→                mainAxisSize: MainAxisSize.min,
   327→                crossAxisAlignment: CrossAxisAlignment.stretch,
   328→                children: [
   329→                  Row(
   330→                    children: [
   331→                      if (icon != null) ...[
   332→                        Icon(icon, color: scheme.primary, size: 22),
   333→                        const SizedBox(width: 10),
   334→                      ],
   335→                      Expanded(
   336→                        child: Text(
   337→                          title,
   338→                          style: Theme.of(sheetContext).textTheme.titleMedium,
   339→                        ),
   340→                      ),
   341→                    ],
   342→                  ),
   343→                  const SizedBox(height: 16),
   344→                  TextField(
   345→                    controller: controller,
   346→                    autofocus: true,
   347→                    keyboardType: const TextInputType.numberWithOptions(
   348→                      decimal: true,
   349→                      signed: false,
   350→                    ),
   351→                    inputFormatters: [
   352→                      FilteringTextInputFormatter.allow(
   353→                        RegExp(r'^\d*[\.,]?\d{0,4}'),
   354→                      ),
   355→                    ],
   356→                    onChanged: (_) {
   357→                      if (error != null) {
   358→                        setSheetState(() => error = null);
   359→                      }
   360→                    },
   361→                    onSubmitted: (_) =>
   362→                        Navigator.of(sheetContext).pop(_parse(controller.text)),
   363→                    decoration: InputDecoration(
   364→                      hintText: hint,
   365→                      errorText: error == null ? null : l10n.sellInvalidNumber,
   366→                    ),
   367→                  ),
   368→                  const SizedBox(height: 18),
   369→                  FilledButton(
   370→                    onPressed: () {
   371→                      final value = _parse(controller.text);
   372→                      if (value == null || value.isNaN || value.isInfinite) {
   373→                        setSheetState(() => error = 'invalid');
   374→                        return;
   375→                      }
   376→                      if (value < 0 || (!allowZero && value == 0)) {
   377→                        setSheetState(() => error = 'invalid');
   378→                        return;
   379→                      }
   380→                      Navigator.of(sheetContext).pop(value);
   381→                    },
   382→                    style: FilledButton.styleFrom(
   383→                      minimumSize: const Size.fromHeight(52),
   384→                    ),
   385→                    child: Text(confirmLabel),
   386→                  ),
   387→                  const SizedBox(height: 8),
   388→                  TextButton(
   389→                    onPressed: () => Navigator.of(sheetContext).pop(),
   390→                    child: Text(l10n.commonCancel),
   391→                  ),
   392→                ],
   393→              ),
   394→            ),
   395→          );
   396→        },
   397→      );
   398→    },
   399→  ).whenComplete(controller.dispose);
   400→}
   401→
   402→double? _parse(String text) {
   403→  final normalized = text.trim().replaceAll(',', '.');
   404→  if (normalized.isEmpty) return null;
   405→  return double.tryParse(normalized);
   406→}

   408→/// محرر خصم (سطر أو رأس) — تبديل نسبة/مبلغ + قيمة، بنتيجة (نوع، قيمة).
   409→Future<(SaleDiscountType, double)?> showDiscountEditSheet(
   410→  BuildContext context, {
   411→  required String title,
   412→  required SaleDiscountType initialType,
   413→  required double initialValue,
   414→}) async {
   415→  var type = initialType;
   416→  final controller = TextEditingController(
   417→    text: initialValue == 0
   418→        ? ''
   419→        : initialValue == initialValue.truncateToDouble()
   420→        ? initialValue.truncate().toString()
   421→        : initialValue.toString(),
   422→  );
   423→  String? error;
   424→  final result = await showModalBottomSheet<(SaleDiscountType, double)>(
   425→    context: context,
   426→    isScrollControlled: true,
   427→    // فوق شريط التبويبات (نفس قرار بقية نوافذ الكاشير).
   428→    useRootNavigator: false,
   429→    builder: (sheetContext) {
   430→      final l10n = AppLocalizations.of(sheetContext)!;
   431→      return StatefulBuilder(
   432→        builder: (sheetContext, setSheetState) {
   433→          return Padding(
   434→            padding: EdgeInsets.only(
   435→              bottom: MediaQuery.of(sheetContext).viewInsets.bottom,
   436→            ),
   437→            child: Padding(
   438→              padding: const EdgeInsets.fromLTRB(20, 20, 20, 24),
   439→              child: Column(
   440→                mainAxisSize: MainAxisSize.min,
   441→                crossAxisAlignment: CrossAxisAlignment.stretch,
   442→                children: [
   443→                  Text(
   444→                    title,
   445→                    style: Theme.of(sheetContext).textTheme.titleMedium,
   446→                  ),
   447→                  const SizedBox(height: 16),
   448→                  SegmentedButton<SaleDiscountType>(
   449→                    segments: [
   450→                      ButtonSegment(
   451→                        value: SaleDiscountType.amount,
   452→                        label: Text(l10n.sellDiscountAmount),
   453→                        icon: const Icon(Icons.sell_outlined),
   454→                      ),
   455→                      ButtonSegment(
   456→                        value: SaleDiscountType.percent,
   457→                        label: Text(l10n.sellDiscountPercent),
   458→                        icon: const Icon(Icons.percent_rounded),
   459→                      ),
   460→                    ],
   461→                    selected: {type},
   462→                    onSelectionChanged: (selection) {
   463→                      setSheetState(() => type = selection.first);
   464→                    },
   465→                  ),
   466→                  const SizedBox(height: 16),
   467→                  TextField(
   468→                    controller: controller,
   469→                    autofocus: true,
   470→                    keyboardType: const TextInputType.numberWithOptions(
   471→                      decimal: true,
   472→                    ),
   473→                    onChanged: (_) {
   474→                      if (error != null) {
   475→                        setSheetState(() => error = null);
   476→                      }
   477→                    },
   478→                    decoration: InputDecoration(
   479→                      suffixText: type == SaleDiscountType.percent ? '%' : null,
   480→                      errorText: error,
   481→                      hintText: type == SaleDiscountType.percent
   482→                          ? l10n.sellDiscountPercentHint
   483→                          : l10n.sellDiscountAmountHint,
   484→                    ),
   485→                  ),
   486→                  const SizedBox(height: 18),
   487→                  FilledButton(
   488→                    onPressed: () {
   489→                      final raw = controller.text.trim().replaceAll(',', '.');
   490→                      final value = raw.isEmpty ? 0.0 : double.tryParse(raw);
   491→                      if (value == null ||
   492→                          value.isNaN ||
   493→                          value < 0 ||
   494→                          (type == SaleDiscountType.percent && value > 100)) {
   495→                        setSheetState(
   496→                          () => error = type == SaleDiscountType.percent
   497→                              ? l10n.sellDiscountPercentError
   498→                              : l10n.sellDiscountAmountError,
   499→                        );
   500→                        return;
   501→                      }
   502→                      Navigator.of(sheetContext).pop((type, value));
   503→                    },
   504→                    style: FilledButton.styleFrom(
   505→                      minimumSize: const Size.fromHeight(52),
   506→                    ),
   507→                    child: Text(l10n.sellDiscountApply),
   508→                  ),
   509→                  const SizedBox(height: 8),
   510→                  TextButton(
   511→                    onPressed: () => Navigator.of(sheetContext).pop(),
   512→                    child: Text(l10n.commonCancel),
   513→                  ),
   514→                ],
   515→              ),
   516→            ),
   517→          );
   518→        },
   519→      );
   520→    },
   521→  );
   522→  controller.dispose();
   523→  return result;
   524→}

   771→/// ينسّق وقتاً 12 ساعة مع ص/م.
   772→String sellFormatTime(DateTime date) {
   773→  final hour = date.hour == 0
   774→      ? 12
   775→      : date.hour > 12
   776→      ? date.hour - 12
   777→      : date.hour;
   778→  final period = date.hour < 12 ? 'ص' : 'م';
   779→  return '${hour.toString().padLeft(2, '0')}:'
   780→      '${date.minute.toString().padLeft(2, '0')} $period';
   781→}
