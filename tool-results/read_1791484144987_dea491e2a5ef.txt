    29→/// قائمة فواتير المبيعات (مسار مقترح `/sell/invoices`).
    30→class SalesInvoicesScreen extends StatelessWidget {
    31→  const SalesInvoicesScreen({super.key, this.viewModel});
    32→
    33→  /// Seam اختبار: نموذج محمّل مسبقاً.
    34→  final SalesInvoicesViewModel? viewModel;
    35→
    36→  @override
    37→  Widget build(BuildContext context) {
    38→    final app = context.read<AppController>();
    39→    late final SalesInvoicesViewModel vm;
    40→    if (viewModel != null) {
    41→      vm = viewModel!;

    44→      unawaited(vm.load());

    50→      child: RefreshOnActive(
    51→        routePattern: RegExp(r'^/sell/invoices$'),
    52→        onActivate: vm.load,
    53→        child: const _SalesInvoicesBody(),
    54→      ),
    55→    );
    56→  }

   117→  Widget _buildContent(BuildContext context, SalesInvoicesViewModel vm) {
   118→    final l10n = AppLocalizations.of(context)!;
   119→    final state = vm.state;
   120→    if (state.loading) {
   121→      return ListView(
   122→        padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   123→        children: const [ListSkeleton(rows: 7)],
   124→      );
   125→    }
   126→    if (state.error != null) {
   127→      return ListView(
   128→        padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   129→        children: [
   130→          ErrorState(
   131→            title: l10n.genericErrorTitle,
   132→            message: l10n.dbOpenErrorMessage,
   133→            technicalDetails: state.error.toString(),
   134→            retryLabel: l10n.commonRetry,
   135→            onRetry: vm.load,
   136→            compact: true,
   137→          ),
   138→        ],
   139→      );
   140→    }
   141→    if (state.invoices.isEmpty) {
   142→      return ListView(
   143→        padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   144→        children: [
   145→          EmptyState(
   146→            icon: Icons.receipt_long_rounded,
   147→            title: l10n.sellInvoicesEmptyTitle,
   148→            message: l10n.sellInvoicesEmptyBody,
   149→            actionLabel: l10n.sellHomeNewInvoice,
   150→            onAction: () => context.go('/sell/new'),
   151→            compact: true,
   152→          ),
   153→        ],
   154→      );
   155→    }
   156→    final visible = state.visible;
   157→    if (visible.isEmpty) {
   158→      return ListView(
   159→        padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   160→        children: [
   161→          EmptyState(
   162→            icon: Icons.search_off_rounded,
   163→            title: l10n.sellInvoicesNoResultsTitle,
   164→            message: l10n.sellInvoicesNoResultsBody,
   165→            compact: true,
   166→          ),
   167→        ],
   168→      );
   169→    }
   170→    return ListView(
   171→      padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   172→      children: [
   173→        for (final invoice in visible)
   174→          Padding(
   175→            padding: const EdgeInsets.only(bottom: 12),
   176→            child: _InvoiceCard(invoice: invoice),
   177→          ),
   178→      ],
   179→    );
   180→  }

   676→/// يُعرض **دائماً** في تفاصيل فاتورة البيع (بلا بوابة صلاحيات). الربح
   677→/// = `total_base − cost_total` — كلا الطرفان بالعملة الأساسية أصلاً
   678→/// (cost_total يُجمَّع من line_cost بالأساس — sale_repository 8-د)،

   694→        ? FinSign.incoming
   695→        : profit < -0.005

   779→/// منازل العملة للطباعة — قرار المستدعي: YER = 0 (قاعدة 5.4-9) والباقي 2.

   781→    detail.currencyCode == 'YER' ? 0 : 2;
   782→
   783→/// يبني إسقاط الطباعة من تفاصيل الفاتورة + رأس المنشأة — كل تسمية داخل
   784→/// المستند مسبقة التعريب من l10n، وكل رقم عبر `AmountText.format`
   785→/// (غربي بفواصل آلاف) فيبقى القالب نفسه بلا أي نص.
