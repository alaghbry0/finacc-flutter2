    30→    30→/// - **سياسة FX المفقود** (FR-08-09 / AC-13): عملة غير أساس بلا سعر اليوم
    31→    31→///   → رفض، إلا إذا فُعّل `fx.fallback=last_known` فيُستخدم آخر سعر مع
    32→    32→///   `rate_is_fallback=1` (شارة FR-02-20) — نفس نمط `sale_repository`.
    33→    33→/// - **اتجاه الصندوق معاكس للبيع** (ملحق و: شراء نقدي CASH → INV): الجزء
    34→    34→///   النقدي سند **صرف** (`tx_type='payment'`) من الصندوق الافتراضي، لا
    35→    35→///   قبض.
    36→    36→///
    37→    37→/// ## قرارات موثقة (انظر worklog-parts/7-b.md للتفصيل):
    38→    38→/// 1. **qty_old لصيغة WAC** = Σ(`stock_level.qty`) للصنف **عبر كل
    39→    39→///    المخازن** داخل المعاملة — التقييم (`product.cost_price`) عالميّ
    40→    40→///    للصنف لا لكل مخزن، فالمتوسط المرجّح يُحسب على المخزون الكلي.
    41→    41→///    والأسطر المتكررة لنفس الصنف تُحدَّث تتابعياً داخل المعاملة نفسها
    42→    42→///    (يُكافئ رياضياً دمجها في تحديث واحد).
    43→    43→/// 2. **الجزء الآجل** (`remainingCredit`) دين للمورد في `invoice.due_amount`
    44→    44→///    حصراً — صيغة رصيد المورد (FR-03-03) تقرأه منه. سند الصرف المصاحب
    45→    45→///    يُكتب بـ `supplier_id = NULL` حتى لا تخصمه صيغة الرصيد (التي تطرح
    46→    46→///    سندات الصرف المرتبطة بالمورد) مرتين — مرآة القرار 2 في
    47→    47→///    `sale_repository`.
    48→    48→/// 3. **payment_allocation** يوثّق الجزء المدفوع نقداً وقت الإصدار (سند
    49→    49→///    الصرف ← الفاتورة) — نفس اصطلاح البيع؛ العمود `cash_tx_id` NOT NULL
    50→    50→///    فلا تخصيص للجزء الآجل بلا سند.
    51→    51→/// 4. **الدفعات الواردة** (FR-01-10): صنف متتبع + رقم دفعة + صلاحية →
    52→    52→///    صف `batch` جديد بتكلفة الوحدة الفعلية **بالعملة الأساسية** (داخل
    53→    53→///    المعاملة مباشرة — `BatchRepository.createBatch` يكتب خارج المعاملة
    54→    54→///    فلا يصلح هنا؛ طلب نسخة تقبل txn مسجَّل للمنسّق). الصنف المتتبع **بلا
    55→    55→///    دفعة** يضاف للمخزون العام بلا دفعة (وفق التكليف) — **تنبيه موثَّق**:
    56→    56→///    فحص توفر البيع للمتتبعين يعدّ الدفعات النشطة فقط، فهذا الرصيد بلا
    57→    57→///    دفعة لا يظهر متاحاً للبيع حتى شراء لاحق بدفعة؛ الواجهة يجب أن تُلزم
    58→    58→///    رقم الدفعة للأصناف المتتبعة. الصنف غير المتتبع مع رقم دفعة → رفض.
    59→    59→/// 5. **رقم فاتورة المورد الورقي** (`supplierInvoiceRef`): لا عمود له في
    60→    60→///    المخطط المجمد — يُخزَّن كبادئة موثَّقة داخل `notes_internal` بصيغة
    61→    61→///    «فاتورة المورد: <الرقم>» (طلب عمود مخصص مسجَّل للمنسّق).
    62→    62→/// 6. **لا دفع زائد**: الشراء لا يعرف «الباقي» — `paidCash` محصور بين
    63→    63→///    0 والصافي (رفض واضح بخلاف البيع الذي يعيد الباقي للعميل).
    64→    64→/// 7. **الأصناف الخدمية** (FR-01-16): تدخل فاتورة الشراء بلا مخزون ولا
    65→    65→///    دفعات ولا WAC — `line_cost = 0` (كلفتها مصروف مستقبلي خارج V1).
    66→    66→/// 8. **المسودة (draft)** خارج نطاق هذا المحرك — البناء للمكتملة فقط
    67→    67→///    (`status='completed'` وفق آلة الحالات 5.4-2).
    68→    68→/// 9. **الضريبة** 0% في V1 لهذا المحرك (`tax_rate=0`) — الأعمدة جاهزة
    69→    69→///    بالمخطط ووحدة الضريبة لاحقة.

   110→   110→/// مستودع المشتريات — ترحيل PUR الذرّي وقراءات فواتير الشراء.
   111→   111→class PurchaseRepository {
   112→   112→  /// يبنى فوق قاعدة مفتوحة؛ يركّب المستودعات الشريكة (تواقيعها ملزمة)
   113→   113→  /// فوق نفس القاعدة دون إعادة إنشاء.
   114→   114→  PurchaseRepository(Database db)
   115→   115→    : _db = db,
   116→   116→      _rates = ExchangeRateRepository(db),
   117→   117→      _settings = SettingsRepository(db);
   118→   118→
   119→   119→  final Database _db;
   120→   120→  final ExchangeRateRepository _rates;
   121→   121→  final SettingsRepository _settings;
   122→   122→
   123→   123→  // ─────────────────────────────────────────────────────────────────────
   124→   124→  // الترحيل الذرّي (FR-02-08 / 5.4-3 / 5.4-4)
   125→   125→  // ─────────────────────────────────────────────────────────────────────
   126→   126→
   127→   127→  /// **ترحيل فاتورة شراء مكتملة** — معاملة واحدة (انظر رأس الملف
   128→   128→  /// للخريطة).
   129→   129→  ///
   130→   130→  /// [userId] منفّذ العملية (يُخزَّن في created_by والتدقيق).
   131→   131→  /// [now] لحظة الكتابة الفعلية (created_at/updated_at/audit) — تاريخ
   132→   132→  /// العمل `draft.issuedAt` وحده يحدد سنة الترقيم وسعر الصرف.
   133→   133→  ///
   134→   134→  /// الترتيب: تحقّق نقّي → تحميلات مسبقة (عملة/سياسة FX/أصناف/مورد/
   135→   135→  /// صندوق) → معاملة واحدة (رقم ← فاتورة ← بنود + WAC + دفعات واردة +
   136→   136→  /// مخزون ← نقدي ← تخصيص ← تدقيق) → إيصال.
   137→   137→  Future<Result<PurchasePostedReceipt, String>> postPurchase(
   138→   138→    PurchaseDraft draft, {
   139→   139→    required int userId,
   140→   140→    DateTime? now,
   141→   141→  }) async {
   142→   142→    final at = now ?? DateTime.now();
   143→   143→    final issuedAt = draft.issuedAt;
   144→   144→    final issuedIso = issuedAt.toUtc().toIso8601String();
   145→   145→
   146→   146→    // (1) التسعير والتحقق النقي (خصومات/كميات/دفع/دفعات واردة) — قبل
   147→   147→    //     أي كتابة.
   148→   148→    final cartFailure = PurchasePricing.validateCart(
   149→   149→      draft.lines,
   150→   150→      invoiceDiscountType: draft.invoiceDiscountType,
   151→   151→      invoiceDiscountValue: draft.invoiceDiscountValue,
   152→   152→    );
   153→   153→    if (cartFailure != null) return Err(cartFailure);
   154→   154→    final priced = PurchasePricing.priceCart(
   155→   155→      draft.lines,
   156→   156→      invoiceDiscountType: draft.invoiceDiscountType,
   157→   157→      invoiceDiscountValue: draft.invoiceDiscountValue,
   158→   158→    );
   159→   159→    final totals = priced.totals;
   160→   160→    final payFailure = PurchasePricing.validatePayment(
   161→   161→      totals.grandTotal,
   162→   162→      draft.paidCash,
   163→   163→      draft.paymentMethod,
   164→   164→    );
   165→   165→    if (payFailure != null) return Err(payFailure);
   166→   166→    final settlement = PurchasePricing.settlePayment(
   167→   167→      totals.grandTotal,
   168→   168→      draft.paidCash,
   169→   169→    );
   170→   170→    final netPaid = settlement.netPaid;
   171→   171→    final remainingCredit = settlement.remainingCredit;
   172→   172→    final payStatus = settlement.payStatus;
   173→   173→
   174→   174→    try {
   175→   175→      // (2) العملة + سياسة FX (FR-08-09) — قراءة خارج المعاملة (نمط
   176→   176→      //     البيع حرفياً).
   177→   177→      final currency = await _db.query(
   178→   178→        'currency',
   179→   179→        columns: ['id', 'code', 'is_base'],
   180→   180→        where: 'id = ?',
   181→   181→        whereArgs: [draft.currencyId],
   182→   182→        limit: 1,
   183→   183→      );
   184→   184→      if (currency.isEmpty) {
   185→   185→        return Err('العملة رقم #${draft.currencyId} غير موجودة.');
   186→   186→      }
   187→   187→      final isBase = (currency.first['is_base'] as int? ?? 0) == 1;
   188→   188→      final currencyCode = currency.first['code'] as String;
   189→   189→      var exchangeRate = 1.0;
   190→   190→      var rateIsFallback = false;
   191→   191→      if (!isBase) {
   192→   192→        final todayRate = await _rates.rateFor(draft.currencyId, issuedAt);
   193→   193→        if (todayRate != null && todayRate > 0) {
   194→   194→          exchangeRate = todayRate;
   195→   195→        } else {
   196→   196→          final fallbackOn =
   197→   197→              (await _settings.getString('fx.fallback', 'off')) == 'last_known';
   198→   198→          if (!fallbackOn) {
   199→   199→            return Err(
   200→   200→              'لا يوجد سعر صرف لعملة $currencyCode بتاريخ اليوم — '
   201→   201→              'أدخل سعر اليوم أولاً ثم احفظ (لا يُحفظ بسعر افتراضي).',
   202→   202→            );
   203→   203→          }
   204→   204→          final lastKnown = await _rates.latestBefore(
   205→   205→            draft.currencyId,
   206→   206→            issuedAt,
   207→   207→          );
   208→   208→          if (lastKnown == null || lastKnown <= 0) {
   209→   209→            return Err(
   210→   210→              'لا يوجد أي سعر صرف معروف لعملة $currencyCode — '
   211→   211→              'أدخل سعراً واحداً على الأقل قبل الشراء بها.',
   212→   212→            );
   213→   213→          }
   214→   214→          exchangeRate = lastKnown;
   215→   215→          rateIsFallback = true; // شارة «سعر صرف تقديري» (FR-02-20).
   216→   216→        }
   217→   217→      }
   218→   218→
   219→   219→      // (3) الأصناف — تحميل واحد مجمّع + رفض المفقود/المؤرشف + فحص
   220→   220→      //     حقول الدفعة الواردة مقابل تتبع الصنف (القرار 4).
   221→   221→      final productIds = draft.lines.map((l) => l.productId).toSet();
   222→   222→      final products = await _loadProducts(productIds);
   223→   223→      for (var i = 0; i < draft.lines.length; i++) {
   224→   224→        final line = draft.lines[i];
   225→   225→        final info = products[line.productId];
   226→   226→        if (info == null) {
   227→   227→          return Err('الصنف رقم #${line.productId} غير موجود.');
   228→   228→        }
   229→   229→        if (info.isArchived) {
   230→   230→          return Err('الصنف «${info.name}» مؤرشف — لا يُشترى (FR-01-15).');
   231→   231→        }
   232→   232→        final batchNo = line.batchNo?.trim() ?? '';
   233→   233→        if (batchNo.isNotEmpty && !info.trackBatches) {
   234→   234→          return Err(
   235→   235→            'الصنف «${info.name}» لا يتتبع الدفعات — احذف رقم الدفعة '
   236→   236→            '«$batchNo» من البند ${i + 1} أو فعّل تتبع الدفعات للصنف.',
   237→   237→          );
   238→   238→        }
   239→   239→      }
   240→   240→
   241→   241→      // (4) المورد (إلزامي للشراء) — وجود + عدم أرشفة.
   242→   242→      final supplierRows = await _db.query(
   243→   243→        'supplier',
   244→   244→        columns: ['name', 'is_archived'],
   245→   245→        where: 'id = ?',
   246→   246→        whereArgs: [draft.supplierId],
   247→   247→        limit: 1,
   248→   248→      );
   249→   249→      if (supplierRows.isEmpty) {
   250→   250→        return Err('المورد رقم #${draft.supplierId} غير موجود.');
   251→   251→      }
   252→   252→      if ((supplierRows.first['is_archived'] as int? ?? 0) == 1) {
   253→   253→        return Err(
   254→   254→          'المورد «${supplierRows.first['name']}» مؤرشف — أزل الأرشفة '
   255→   255→          'أولاً أو اختر مورداً آخر.',
   256→   256→        );
   257→   257→      }
   258→   258→
   259→   259→      // (5) الصندوق الافتراضي للمنشأة — للجزء النقدي فقط.
   260→   260→      int? cashboxId;
   261→   261→      if (netPaid > moneyEpsilon) {
   262→   262→        final boxRows = await _db.query(
   263→   263→          'cashbox',
   264→   264→          columns: ['id'],
   265→   265→          where: 'is_default = 1 AND is_archived = 0',
   266→   266→          limit: 1,
   267→   267→        );
   268→   268→        if (boxRows.isEmpty) {
   269→   269→          return Err('لا يوجد صندوق افتراضي للمنشأة — أنشئ صندوقاً أولاً.');
   270→   270→        }
   271→   271→        cashboxId = boxRows.first['id'] as int;
   272→   272→      }
   273→   273→
   274→   274→      // (6) المعاملة الواحدة — كل الكتابات أو لا شيء (5.4-4).
   275→   275→      return await _db.transaction((txn) async {
   276→   276→        // 6-أ) رقم PUR ذرّي داخل المعاملة نفسها (قاعدة 5.4-1).
   277→   277→        final year = issuedAt.year; // سنة يوم العمل المحلية.
   278→   278→        final seq = DocSequenceService(txn);
   279→   279→        final number = await seq.nextNumber(DocSequenceType.purchase, year);
   280→   280→        final docNo = formatDocNumber(DocSequenceType.purchase, year, number);
   281→   281→
   282→   282→        // 6-ب) رأس الفاتورة (كل الأعمدة وفق §5.3) — رقم فاتورة المورد
   283→   283→        //      الورقي كبادئة موثَّقة داخل الملاحظة الداخلية (القرار 5).
   284→   284→        final totalBase = roundCost(totals.grandTotal * exchangeRate);
   285→   285→        final notesInternal = _composeInternalNotes(
   286→   286→          draft.supplierInvoiceRef,
   287→   287→          draft.notesInternal,
   288→   288→        );
   289→   289→        final invoiceId = await txn.insert('invoice', {
   290→   290→          'invoice_no': docNo,
   291→   291→          'doc_type': 'purchase',
   292→   292→          'pay_status': payStatus.code,
   293→   293→          'status': 'completed',
   294→   294→          'issued_at': issuedIso,
   295→   295→          'supplier_id': draft.supplierId,
   296→   296→          'cashbox_id': cashboxId,
   297→   297→          'warehouse_id': draft.warehouseId,
   298→   298→          'currency_id': draft.currencyId,
   299→   299→          'exchange_rate': exchangeRate,
   300→   300→          'rate_is_fallback': rateIsFallback ? 1 : 0,
   301→   301→          'subtotal': totals.subtotal,
   302→   302→          'discount_amount': roundMoney(totals.totalDiscount),
   303→   303→          'tax_rate': 0,
   304→   304→          'tax_amount': 0,
   305→   305→          'total': totals.grandTotal,
   306→   306→          'total_base': totalBase,
   307→   307→          'paid_amount': netPaid,
   308→   308→          'due_amount': remainingCredit,
   309→   309→          'cost_total': 0, // يُجمَّع من الأسطر ثم يُحدَّث أدناه.
   310→   310→          'notes_internal': notesInternal,
   311→   311→          'notes_printed': draft.notesPrinted,
   312→   312→          'created_at': at.toUtc().toIso8601String(),
   313→   313→          'updated_at': at.toUtc().toIso8601String(),
   314→   314→          'created_by': userId,
   315→   315→        });
   316→   316→
   317→   317→        // 6-ج) الأسطر: تكلفة الوحدة الفعلية بالعملة الأساسية ÷ المستلم
   318→   318→        //      الكلي (بونص R16-a) → WAC → الدفعة الواردة → السطر →
   319→   319→        //      المخزون وحركته — سطراً سطراً.
   320→   320→        var costTotal = 0.0;
   321→   321→        for (final pricedLine in priced.lines) {
   322→   322→          final info = products[pricedLine.line.productId]!;
   323→   323→          final lineQty = pricedLine.line.qty;
   324→   324→          // المستلم الكلي (بونص R16-a): المدفوع + المجاني يدخلان المخزون
   325→   325→          // معاً — > 0 دائماً لأن qty > 0 مُتحقَّق في PurchasePricing.
   326→   326→          final freeQty = pricedLine.line.freeQty;
   327→   327→          final receivedQty = lineQty + freeQty;
   328→   328→
   329→   329→          // تكلفة الوحدة الفعلية بالعملة الأساسية بسعر يوم الشراء (AC-03)
   330→   330→          // — **بالقسمة على المستلم الكلي** (قرار تحاسبي موثّق R16-a:
   331→   331→          // إجمالي التكلفة ÷ إجمالي الكمية المستلمة — البونص يخفّض
   332→   332→          // التكلفة الوحدوية: 10+2 مجاني بتكلفة 1200 → 100)؛ الخدمي بلا
   333→   333→          // تكلفة مخزونية (القرار 7).
   334→   334→          final unitCostBase = info.isService
   335→   335→              ? 0.0
   336→   336→              : roundCost(pricedLine.netFinal * exchangeRate / receivedQty);
   337→   337→          final lineCost = roundCost(receivedQty * unitCostBase);
   338→   338→          costTotal = roundCost(costTotal + lineCost);
   339→   339→
   340→   340→          // WAC (5.4-3): يُقرأ ويُحدَّث داخل المعاملة على المستلم الكلي —
   341→   341→          // قلب الشراء (بونص R16-a: الكمية الجديدة بالكلي).
   342→   342→          if (!info.isService) {
   343→   343→            await _applyWac(
   344→   344→              txn,
   345→   345→              productId: info.id,
   346→   346→              qtyNew: receivedQty,
   347→   347→              costNew: unitCostBase,
   348→   348→              now: at,
   349→   349→            );
   350→   350→          }
   351→   351→
   352→   352→          // الدفعة الواردة (القرار 4): متتبع + رقم + صلاحية → صف batch
   353→   353→          // داخل المعاملة مباشرة (createBatch يكتب خارجها فلا يصلح).
   354→   354→          // بونص R16-a: الدفعة تستقبل الكلي بتكلفة الوحدة الوحدوية.
   355→   355→          int? batchId;

   594→   594→  /// **WAC (قاعدة 5.4-3)** — يقرأ التكلفة والكمية الكلية داخل المعاملة
   595→   595→  /// ثم يحدّث `product.cost_price` بدقة أربع منازل:
   596→   596→  ///
   597→   597→  /// - `qty_old > 0`:
   598→   598→  ///   `new = (qty_old×cost_old + qty_new×cost_new)/(qty_old + qty_new)`
   599→   599→  /// - `qty_old ≤ 0`: `new = cost_new` مباشرة (لا متوسط على مخزون معدوم).
   600→   600→  ///
   601→   601→  /// `qty_old` = Σ stock_level.qty عبر كل المخازن (القرار 1 بالرأس).
   602→   602→  Future<void> _applyWac(
   603→   603→    DatabaseExecutor txn, {
   604→   604→    required int productId,
   605→   605→    required double qtyNew,
   606→   606→    required double costNew,
   607→   607→    required DateTime now,
   608→   608→  }) async {
   609→   609→    final rows = await txn.rawQuery(
   610→   610→      'SELECT p.cost_price AS cost, '
   611→   611→      '       COALESCE((SELECT SUM(sl.qty) FROM stock_level sl '
   612→   612→      '                 WHERE sl.product_id = p.id), 0) AS qty '
   613→   613→      'FROM product p WHERE p.id = ?',
   614→   614→      [productId],
   615→   615→    );
   616→   616→    if (rows.isEmpty) {
   617→   617→      throw _FlowError('الصنف رقم #$productId غير موجود.');
   618→   618→    }
   619→   619→    final costOld = (rows.first['cost'] as num?)?.toDouble() ?? 0;
   620→   620→    final qtyOld = (rows.first['qty'] as num?)?.toDouble() ?? 0;
   621→   621→    double newCost;
   622→   622→    if (qtyOld <= _qtyEpsilon) {
   623→   623→      newCost = costNew; // المخزون معدوم/سالب دفترياً → التكلفة الجديدة.
   624→   624→    } else {
   625→   625→      newCost = roundCost(
   626→   626→        (qtyOld * costOld + qtyNew * costNew) / (qtyOld + qtyNew),
   627→   627→      );
   628→   628→    }
   629→   629→    if (newCost < 0) newCost = 0; // حارس نظري (لا تكلفة سالبة).
   630→   630→    await txn.update(
   631→   631→      'product',
   632→   632→      {'cost_price': newCost, 'updated_at': now.toUtc().toIso8601String()},
   633→   633→      where: 'id = ?',
   634→   634→      whereArgs: [productId],
   635→   635→    );
   636→   636→  }

   672→   672→  String _describeDbError(DatabaseException e) {
   673→   673→    if (e.isUniqueConstraintError()) {

   692→   692→    }

   694→   694→  }

   697→   697→/// هل الخطأ خرقاً لقيد CHECK؟ (لا مساعد جاهز في sqflite — نقرأ الرسالة).
   698→   698→bool _isCheckFailure(DatabaseException e) =>
   699→   699→    e.toString().toUpperCase().contains('CHECK');
   700→   700→
   701→   701→/// تفاوت الكميات (NUMERIC(12,3) بالمخطط).
   702→   702→const double _qtyEpsilon = 0.000001;
   703→   703→
   704→   704→/// يصيغ رقماً للعرض في الرسائل بلا أصفار زائدة.
   705→   705→String _num(double v) {
   706→   706→  if (v.isNaN || v.isInfinite) {
   707→   707→    return v.toString();
   708→   708→  }
   709→   709→  final rounded = (v * 1000).round() / 1000;
   710→   710→  if (rounded == rounded.round()) {
   711→   711→    return rounded.toInt().toString();
   712→   712→  }
   713→   713→  return rounded.toStringAsFixed(3);
   714→   714→}

   716→   716→/// `YYYY-MM-DD` بتاريخ التقويم المحلي (صيغة `batch.expiry_date` الموحدة).
   717→   717→String _dateOnly(DateTime d) =>
   718→   718→    '${d.year.toString().padLeft(4, '0')}-'
   719→   719→    '${d.month.toString().padLeft(2, '0')}-'
   720→   720→    '${d.day.toString().padLeft(2, '0')}';
   721→   721→
