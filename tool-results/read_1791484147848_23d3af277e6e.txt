    32→/// تاريخ `يوم/شهر/سنة` بأرقام بسيطة (نمط شاشة الوردية).
    33→String stocktakeFormatDate(DateTime date) {
    34→  final day = date.day.toString().padLeft(2, '0');
    35→  final month = date.month.toString().padLeft(2, '0');
    36→  return '$day/$month/${date.year}';
    37→}

    39→/// كمية بثلاث منازل (NUMERIC(12,3)) — نص عربيّ بلا فواصل آلاف للحقول.
    40→String stocktakeQty(double value) {
    41→  final rounded = (value * 1000).round() / 1000;
    42→  final text = rounded == rounded.truncateToDouble()
    43→      ? rounded.truncate().toString()
    44→      : rounded.toStringAsFixed(3);
    45→  return text;
    46→}

   491→    final cleaned = raw.trim().replaceAll(',', '.');
   492→    if (cleaned.isEmpty) {
   493→      vm.setCounted(widget.line.productId, null);
   494→      return;
   495→    }
   496→    final value = double.tryParse(cleaned);
   497→    if (value == null || value.isNaN || value.isInfinite || value < 0) {
   498→      vm.setCounted(widget.line.productId, null);
   499→      return;
   500→    }
   501→    vm.setCounted(widget.line.productId, value);
