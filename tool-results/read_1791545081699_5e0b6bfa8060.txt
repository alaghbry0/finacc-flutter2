     1→/// محرك ترحيل فاتورة البيع الذرّي — جداول `invoice` / `invoice_item` /
     2→/// `stock_level` / `stock_movement` / `batch` / `cash_tx` /
     3→/// `payment_allocation` / `quotation` — المرحلة 4 (FR-02 / FR-08-09).
     4→///
     5→/// **الخريطة الملزمة** (قاعدة 5.4-4 + ملحق و): الحفظ = (رقم INV ذرّي +
     6→/// الفاتورة + بنودها بـ line_cost=WAC + خصم المخزون مع FEFO + حركات
     7→/// المخزون + حركة الصندوق للجزء النقدي + تخصيص المدفوع + تعليم عرض
     8→/// السعر المحوَّل + قيد التدقيق) داخل **Transaction واحدة** — فشل أي خطوة
     9→/// يرجع الكل (AC-09-أ).
    10→///
    11→/// خطوط حمراء مطبَّقة هنا:
    12→/// - **منع السالب المخزوني** (5.4-5): فحص مسبق قبل المعاملة + حارس
    13→///   `WHERE qty >= ?` داخلها — رسالة الرفض تسمّي الصنف والكمية المتاحة.
    14→/// - **فصل العملات** (5.4-7): الفاتورة وسندها النقدي وتخصيصها كلها بعملة
    15→///   الفاتورة وسعر يومها (Snapshot FR-08-05)؛ الأساس بالعملة الأساسية
    16→///   يُخزَّن في `total_base` فقط.
    17→/// - **WAC** (5.4-3): البيع **يستهلك** `product.cost_price` ويخزنه في
    18→///   `line_cost` ولا يغيّره أبداً.
    19→/// - **سياسة FX المفقود** (FR-08-09 / AC-13): عملة غير أساس بلا سعر اليوم
    20→///   → رفض، إلا إذا فُعّل `fx.fallback=last_known` فيُستخدم آخر سعر مع
    21→///   `rate_is_fallback=1` (شارة FR-02-20).
    22→///
    23→/// ## قرارات موثقة (انظر worklog-parts/6-b.md للتفصيل):
    24→/// 1. **الباقي للعميل** (`changeDue`): الزيادة النقدية فوق الصافي تُعاد في
    25→///    الإيصال ولا تدخل الصندوق (`cash_tx.amount` = الصافي المدفوع حصراً
    26→///    لأن عموده CHECK(amount > 0) ولا معنى لسند بمبلغ الباقي) ولا الدين.
    27→/// 2. **payment_allocation**: العمود `cash_tx_id` NOT NULL فلا يمكن إدراجه
    28→///    لجزء آجل بلا سند — التخصيص يوثّق **الجزء المدفوع نقداً** وقت
    29→///    الإصدار (سند القبض ← الفاتورة)، والأثر الآجل محفوظ في
    30→///    `invoice.due_amount` وفق اصطلاح CustomerRepository (لا يُنقصه أحد).
    31→///    سند قبض الإصدار يُكتب بـ `customer_id = NULL` حتى لا تخصمه صيغة
    32→///    رصيد العميل (FR-03-02) مرتين.
    33→/// 3. **دفع متعدد الدفعات**: سطر فاتورة واحد لكل سطر سلة؛ تفصيل الدفعات
    34→///    يُسجَّل حركة مخزون مستقلة لكل دفعة (راجعة بالفاتورة) +
    35→///    `invoice_item.batch_id` = أول دفعة FEFO وملخص الدفعات في ملاحظة
    36→///    السطر (AC-05: «أسطر الفاتورة أظهرت رقم الدفعة»).
    37→/// 4. **المسودة (draft)** خارج نطاق هذا المحرك — البناء هنا للمكتملة فقط
    38→///    (`status='completed'` وفق آلة الحالات 5.4-2).
    39→/// 5. **الضريبة** 0% في V1 لهذا المحرك (`tax_rate=0`) — الأعمدة جاهزة
    40→///    بالمخطط ووحدة الضريبة لاحقة.
    41→/// 6. **userId** يُمرَّر صراحة (لا مساعد مستخدم حالي في طبقة البيانات —
    42→///    الطلب مسجَّل في التسليم للمنسّق).
    43→/// 7. **البونص/الكميات المجانية** (موجة UX-4 — القرار التحاسبي الملزم
    44→///    من UX-audit-invoice §3): `CartLine.freeQty` ≥ 0 عمود مستقل
    45→///    `invoice_item.free_qty` (هجرة v5). **المنصرف الكلي = qty + freeQty**
    46→///    (فحص التوفر + FEFO + `line_cost = roundCost(الكل×WAC)` + حركة
    47→///    المخزون −الكل + stock_level)، و**الإيراد من qty حصراً** —
    48→///    subtotal/total/total_base/الائتمان لا ترى البونص إطلاقاً (تسعير
    49→///    `SalePricing` فوق qty وحدها)، فCOGS يرتفع والربح ينخفض بالمقدار
    50→///    الحرفي للمجاني. حركة المخزون وملاحظة السطر تحملان «بونص: N»
    51→///    والتدقيق يسجل `free=`. رسائل العجز تشمل المجاني بالمطلوب.

    97→/// مستودع المبيعات — ترحيل POS الذرّي وقراءات الفواتير.
    98→class SaleRepository {
    99→  /// يبنى فوق قاعدة مفتوحة؛ يركّب المستودعات الشريكة (تواقيعها ملزمة)
   100→  /// فوق نفس القاعدة دون إعادة إنشاء.
   101→  SaleRepository(Database db)
   102→    : _db = db,
   103→      _rates = ExchangeRateRepository(db),
   104→      _settings = SettingsRepository(db),
   105→      _customers = CustomerRepository(db),
   106→      _batches = BatchRepository(db);
   107→
   108→  final Database _db;
   109→  final ExchangeRateRepository _rates;
   110→  final SettingsRepository _settings;
   111→  final CustomerRepository _customers;
   112→  final BatchRepository _batches;
   113→
   114→  // ─────────────────────────────────────────────────────────────────────
   115→  // الترحيل الذرّي (FR-02-06 / 5.4-4)
   116→  // ─────────────────────────────────────────────────────────────────────
   117→
   118→  /// **ترحيل فاتورة بيع مكتملة** — معاملة واحدة (انظر رأس الملف للخريطة).
   119→  ///
   120→  /// [userId] منفّذ العملية (يُخزَّن في created_by والتدقيق).
   121→  /// [now] لحظة الكتابة الفعلية (created_at/updated_at/audit) — تاريخ
   122→  /// العمل `draft.issuedAt` وحده يحدد سنة الترقيم وسعر الصرف.
   123→  ///
   124→  /// الترتيب: تحقّق نقّي → تحميلات مسبقة (عملة/سياسة FX/أصناف/مخزون/
   125→  /// ائتمان/صندوق) → معاملة واحدة (رقم ← فاتورة ← بنود + WAC + FEFO +
   126→  /// مخزون ← نقدي ← تخصيص ← ربط عرض السعر ← تدقيق) → إيصال.
   127→  Future<Result<SalePostedReceipt, String>> postSale(
   128→    SaleDraft draft, {
   129→    required int userId,
   130→    DateTime? now,
   131→  }) async {
   132→    final at = now ?? DateTime.now();
   133→    final issuedAt = draft.issuedAt;
   134→    final issuedIso = issuedAt.toUtc().toIso8601String();
   135→
   136→    // (1) التسعير والتحقق النقي (خصومات/كميات/دفع) — قبل أي كتابة.
   137→    final cartFailure = SalePricing.validateCart(
   138→      draft.lines,
   139→      invoiceDiscountType: draft.invoiceDiscountType,
   140→      invoiceDiscountValue: draft.invoiceDiscountValue,
   141→    );
   142→    if (cartFailure != null) return Err(cartFailure);
   143→    final priced = SalePricing.priceCart(
   144→      draft.lines,
   145→      invoiceDiscountType: draft.invoiceDiscountType,
   146→      invoiceDiscountValue: draft.invoiceDiscountValue,
   147→    );
   148→    final totals = priced.totals;
   149→    final payFailure = SalePricing.validatePayment(
   150→      totals.grandTotal,
   151→      draft.paidCash,
   152→      draft.paymentMethod,
   153→    );
   154→    if (payFailure != null) return Err(payFailure);
   155→    final settlement = SalePricing.settlePayment(
   156→      totals.grandTotal,
   157→      draft.paidCash,
   158→    );
   159→    final netPaid = settlement.netPaid;
   160→    final changeDue = settlement.changeDue;
   161→    final remainingCredit = settlement.remainingCredit;
   162→    final payStatus = settlement.payStatus;
   163→
   164→    try {
   165→      // (2) العملة + سياسة FX (FR-08-09) — قراءة خارج المعاملة.
   166→      final currency = await _db.query(
   167→        'currency',
   168→        columns: ['id', 'code', 'is_base'],
   169→        where: 'id = ?',
   170→        whereArgs: [draft.currencyId],
   171→        limit: 1,
   172→      );
   173→      if (currency.isEmpty) {
   174→        return Err('العملة رقم #${draft.currencyId} غير موجودة.');
   175→      }
   176→      final isBase = (currency.first['is_base'] as int? ?? 0) == 1;
   177→      final currencyCode = currency.first['code'] as String;
   178→      var exchangeRate = 1.0;
   179→      var rateIsFallback = false;
   180→      if (!isBase) {
   181→        final todayRate = await _rates.rateFor(draft.currencyId, issuedAt);
   182→        if (todayRate != null && todayRate > 0) {
   183→          exchangeRate = todayRate;
   184→        } else {
   185→          final fallbackOn =
   186→              (await _settings.getString('fx.fallback', 'off')) == 'last_known';
   187→          if (!fallbackOn) {
   188→            return Err(
   189→              'لا يوجد سعر صرف لعملة $currencyCode بتاريخ اليوم — '
   190→              'أدخل سعر اليوم أولاً ثم احفظ (لا يُحفظ بسعر افتراضي).',
   191→            );
   192→          }
   193→          final lastKnown = await _rates.latestBefore(
   194→            draft.currencyId,
   195→            issuedAt,
   196→          );
   197→          if (lastKnown == null || lastKnown <= 0) {
   198→            return Err(
   199→              'لا يوجد أي سعر صرف معروف لعملة $currencyCode — '
   200→              'أدخل سعراً واحداً على الأقل قبل البيع بها.',
   201→            );
   202→          }
   203→          exchangeRate = lastKnown;
   204→          rateIsFallback = true; // شارة «سعر صرف تقديري» (FR-02-20).
   205→        }
   206→      }
   207→
   208→      // (3) الأصناف — تحميل واحد مجمّع + رفض المفقود/المؤرشف.
   209→      final productIds = draft.lines.map((l) => l.productId).toSet();
   210→      final products = await _loadProducts(productIds);
   211→      for (final id in productIds) {
   212→        final info = products[id];
   213→        if (info == null) {
   214→          return Err('الصنف رقم #$id غير موجود.');
   215→        }
   216→        if (info.isArchived) {
   217→          return Err('الصنف «${info.name}» مؤرشف — لا يُباع (FR-01-15).');
   218→        }
   219→      }
   220→
   221→      // (4) فحص المخزون المسبق (5.4-5): لكل صنف غير خدمي — المتاح من
   222→      //     stock_level (العادي) أو دفعات FEFO النشطة (المتتبع).
   223→      final stockFailure = await _checkAvailability(
   224→        draft.lines,
   225→        products,
   226→        draft.warehouseId,
   227→        issuedAt,
   228→      );
   229→      if (stockFailure != null) return Err(stockFailure);
   230→
   231→      // (5) العميل + حد الائتمان (FR-03-05) — الآجل فقط.
   232→      if (draft.customerId != null) {
   233→        final customerRows = await _db.query(
   234→          'customer',
   235→          columns: ['name', 'is_archived'],
   236→          where: 'id = ?',
   237→          whereArgs: [draft.customerId],
   238→          limit: 1,
   239→        );
   240→        if (customerRows.isEmpty) {
   241→          return Err('العميل رقم #${draft.customerId} غير موجود.');
   242→        }
   243→        if ((customerRows.first['is_archived'] as int? ?? 0) == 1) {
   244→          return Err(
   245→            'العميل «${customerRows.first['name']}» مؤرشف — '
   246→            'أزل الأرشفة أولاً أو اختر عميلاً آخر.',
   247→          );
   248→        }
   249→        if (remainingCredit > moneyEpsilon) {
   250→          final credit = await _customers.checkCredit(
   251→            draft.customerId!,
   252→            draft.currencyId,
   253→            remainingCredit,
   254→          );
   255→          if (credit.overLimit) {
   256→            // FR-03-05 (17-c): السلوك من إعداد parties.credit_limit_action
   257→            // — 'block' يرفض هنا نهائياً؛ 'warn' (الافتراضي) حواره في
   258→            // PaymentSheet («متابعة على أي حال») وقد أقرّه المستخدم،
   259→            // فلا يمنع الترحيل — هذا المستودع حارس 'block' حصراً.
   260→            final action = await _settings.getString(
   261→              'parties.credit_limit_action',
   262→              'warn',
   263→            );
   264→            if (action == 'block') {
   265→              return Err(
   266→                'تجاوز حد الائتمان: رصيد العميل الحالي ${_num(credit.balance)} '
   267→                '+ الآجل الجديد ${_num(remainingCredit)} يتجاوز الحد '
   268→                '${_num(credit.creditLimit ?? 0)} — قلّل الآجل أو استوفِ أولاً.',
   269→              );
   270→            }
   271→          }
   272→        }
   273→      }
   274→
   275→      // (6) الصندوق الافتراضي للمنشأة — للجزء النقدي فقط.
   276→      int? cashboxId;
   277→      if (netPaid > moneyEpsilon) {
   278→        final boxRows = await _db.query(
   279→          'cashbox',
   280→          columns: ['id'],
   281→          where: 'is_default = 1 AND is_archived = 0',
   282→          limit: 1,
   283→        );
   284→        if (boxRows.isEmpty) {
   285→          return Err('لا يوجد صندوق افتراضي للمنشأة — أنشئ صندوقاً أولاً.');
   286→        }
   287→        cashboxId = boxRows.first['id'] as int;
   288→      }
   289→
   290→      // (7) عرض السعر المصدر — تحقق مسبق (الحراسة النهائية داخل المعاملة).
   291→      if (draft.fromQuotationId != null) {
   292→        final quoteRows = await _db.query(
   293→          'quotation',
   294→          columns: ['status'],
   295→          where: 'id = ?',
   296→          whereArgs: [draft.fromQuotationId],
   297→          limit: 1,
   298→        );
   299→        if (quoteRows.isEmpty) {
   300→          return Err('عرض السعر رقم #${draft.fromQuotationId} غير موجود.');
   301→        }
   302→        final status = quoteRows.first['status'] as String;
   303→        if (status != 'draft' && status != 'sent') {
   304→          return Err('عرض السعر غير قابل للتحويل — حالته الحالية «$status».');
   305→        }
   306→      }
   307→
   308→      // (8) المعاملة الواحدة — كل الكتابات أو لا شيء (5.4-4).
   309→      return await _db.transaction((txn) async {
   310→        // 8-أ) رقم INV ذرّي داخل المعاملة نفسها (قاعدة 5.4-1).
   311→        final year = issuedAt.year; // سنة يوم العمل المحلية.
   312→        final seq = DocSequenceService(txn);
   313→        final number = await seq.nextNumber(DocSequenceType.invoice, year);
   314→        final invoiceNo = formatDocNumber(
   315→          DocSequenceType.invoice,
   316→          year,
   317→          number,
   318→        );
   319→
   320→        // 8-ب) رأس الفاتورة (كل الأعمدة وفق §5.3).
   321→        final totalBase = roundCost(totals.grandTotal * exchangeRate);
   322→        final invoiceId = await txn.insert('invoice', {
   323→          'invoice_no': invoiceNo,
   324→          'doc_type': 'sale',
   325→          'pay_status': payStatus.code,
   326→          'status': 'completed',
   327→          'issued_at': issuedIso,
   328→          'quotation_id': draft.fromQuotationId,
   329→          'customer_id': draft.customerId,
   330→          'cashbox_id': cashboxId,
   331→          'warehouse_id': draft.warehouseId,
   332→          'currency_id': draft.currencyId,
   333→          'exchange_rate': exchangeRate,
   334→          'rate_is_fallback': rateIsFallback ? 1 : 0,
   335→          'subtotal': totals.subtotal,
   336→          'discount_amount': roundMoney(totals.totalDiscount),
   337→          'tax_rate': 0,
   338→          'tax_amount': 0,
   339→          'total': totals.grandTotal,
   340→          'total_base': totalBase,
   341→          'paid_amount': netPaid,
   342→          'due_amount': remainingCredit,
   343→          'cost_total': 0, // يُجمَّع من الأسطر ثم يُحدَّث أدناه.
   344→          'notes_internal': draft.notesInternal,
   345→          'notes_printed': draft.notesPrinted,
   346→          'created_at': at.toUtc().toIso8601String(),
   347→          'updated_at': at.toUtc().toIso8601String(),
   348→          'created_by': userId,
   349→        });
   350→
   351→        // 8-ج) الأسطر: WAC داخل المعاملة قبل الاستهلاك (5.4-3) + FEFO +
   352→        //      المخزون + حركاته — سطراً سطراً. البونص (UX-4): المنصرف
   353→        //      الكلي = المدفوع + المجاني على المخزون/الدفعات/التكلفة.
   354→        var costTotal = 0.0;
   355→        var freeTotal = 0.0;
   356→        for (var i = 0; i < priced.lines.length; i++) {
   357→          final pricedLine = priced.lines[i];
   358→          final info = products[pricedLine.line.productId]!;
   359→          final lineQty = pricedLine.line.qty;
   360→          final freeQty = pricedLine.line.freeQty;
   361→          freeTotal += freeQty;
   362→          // المنصرف الكلي (القرار 7) — البونص وحدات حقيقية تخرج من
   363→          // المخزون وتكلفتها تدخل COGS.
   364→          final dispatchedQty = lineQty + freeQty;
   365→
   366→          // WAC الحالي يُقرأ داخل المعاملة (Snapshot لحظة البيع) — على
   367→          // المنصرف الكلي.
   368→          final costNow = await _readCostInside(txn, info.id);
   369→          final lineCost = info.isService
   370→              ? 0.0
   371→              : roundCost(dispatchedQty * costNow);
   372→          costTotal = roundCost(costTotal + lineCost);
   373→
   374→          // تخصيص FEFO للصنوف المتتبعة (AC-05) قبل كتابة السطر — على
   375→          // المنصرف الكلي.
   376→          FefoResult? fefo;
   377→          if (!info.isService && info.trackBatches) {
   378→            fefo = await _batches.allocateFefo(
   379→              txn,
   380→              productId: info.id,
   381→              warehouseId: draft.warehouseId,
   382→              qty: dispatchedQty,
   383→              asOf: issuedAt,
   384→            );
   385→            if (fefo.shorted) {
   386→              throw _FlowError(
   387→                'الكمية غير متوفرة للصنف «${info.name}»: المتاح '
   388→                '${_num(fefo.allocatedQty)} والمطلوب ${_num(dispatchedQty)}'
   389→                '${freeQty > _qtyEpsilon ? ' (منها ${_num(freeQty)} مجاني)' : ''} — '
   390→                'لا يسمح النظام بمخزون سالب.',
   391→              );
   392→            }
   393→          }
   394→
   395→          final batchSummary = fefo == null || fefo.allocations.isEmpty
   396→              ? null
   397→              : fefo.allocations
   398→                    .map((a) => '${a.batchNumber}×${_num(a.qty)}')
   399→                    .join('، ');
   400→          final lineNotes = [
   401→            if ((pricedLine.line.notes ?? '').trim().isNotEmpty)
   402→              pricedLine.line.notes!.trim(),
   403→            if (freeQty > _qtyEpsilon) 'بونص: ${_num(freeQty)}',
   404→            if (batchSummary != null) 'دفعة: $batchSummary',
   405→          ].join(' — ');
   406→
   407→          await txn.insert('invoice_item', {
   408→            'invoice_id': invoiceId,
   409→            'product_id': info.id,
   410→            'line_desc': info.name, // لقطة اسم الصنف للعرض/الطباعة.
   411→            'qty': lineQty,
   412→            'free_qty': freeQty, // البونص مستقل (UX-4 — القرار 7).
   413→            'unit_id': info.unitId,
   414→            'unit_factor': 1,
   415→            'unit_price': pricedLine.line.unitPrice,
   416→            'discount_percent':
   417→                pricedLine.line.lineDiscountType == SaleDiscountType.percent
   418→                ? pricedLine.line.lineDiscountValue
   419→                : 0,
   420→            'discount_amount': pricedLine.effectiveDiscount,
   421→            'tax_percent': 0,
   422→            'line_total': pricedLine.netFinal,
   423→            'line_cost': lineCost,
   424→            'batch_id': (fefo != null && fefo.allocations.isNotEmpty)
   425→                ? fefo.allocations.first.batchId
   426→                : null,
   427→            'notes': lineNotes.isEmpty ? null : lineNotes,
   428→            'created_at': at.toUtc().toIso8601String(),
   429→          });
   430→
   431→          // الأصناف الخدمية: لا مخزون ولا دفعات (التكلفة صفر أعلاه).
   432→          if (info.isService) continue;
   433→
   434→          if (fefo != null) {
   435→            // المتتبع: تطبيق الحصص على الدفعات (حارس السالب بداخلها
   436→            // يرمي فتتراجع المعاملة كاملة) + حركة لكل دفعة.
   437→            await _batches.applyAllocation(txn, fefo.allocations, now: at);
   438→            for (final allocation in fefo.allocations) {
   439→              await txn.insert('stock_movement', {
   440→                'product_id': info.id,
   441→                'warehouse_id': draft.warehouseId,
   442→                'movement_type': 'sale',
   443→                'qty': -allocation.qty, // الخروج سالب (اتجاه المخطط).
   444→                'unit_cost': costNow,
   445→                'ref_type': 'invoice',
   446→                'ref_id': invoiceId,
   447→                'moved_at': issuedIso,
   448→                'notes':
   449→                    'رقم الدفعة ${allocation.batchNumber} (تنتهي '
   450→                    '${_dateOnly(allocation.expiryDate)})'
   451→                    '${freeQty > _qtyEpsilon ? ' — بونص: ${_num(freeQty)}' : ''}',
   452→                'created_at': at.toUtc().toIso8601String(),
   453→                'created_by': userId,
   454→              });
   455→            }
   456→          } else {
   457→            // العادي: حركة واحدة للسطر — بمنصرفه الكلي وبوسم البونص.
   458→            await txn.insert('stock_movement', {
   459→              'product_id': info.id,
   460→              'warehouse_id': draft.warehouseId,
   461→              'movement_type': 'sale',
   462→              'qty': -dispatchedQty,
   463→              'unit_cost': costNow,
   464→              'ref_type': 'invoice',
   465→              'ref_id': invoiceId,
   466→              'moved_at': issuedIso,
   467→              'notes': freeQty > _qtyEpsilon
   468→                  ? '$invoiceNo — بونص: ${_num(freeQty)}'
   469→                  : invoiceNo,
   470→              'created_at': at.toUtc().toIso8601String(),
   471→              'created_by': userId,
   472→            });
   473→          }
   474→
   475→          // دفتر stock_level مشترك للنوعين — حارس السالب الصارم:
   476→          // (INSERT OR IGNORE يهيئ صفاً معدوماً ثم الشرط qty >= ? يمنع
   477→          // أي خصم تحت الصفر — بلا استثناء).
   478→          await txn.rawInsert(
   479→            'INSERT OR IGNORE INTO stock_level(product_id, warehouse_id, qty) '
   480→            'VALUES(?, ?, 0)',
   481→            [info.id, draft.warehouseId],
   482→          );
   483→          final consumed = await txn.rawUpdate(
   484→            'UPDATE stock_level SET qty = qty - ? '
   485→            'WHERE product_id = ? AND warehouse_id = ? AND qty >= ?',
   486→            [dispatchedQty, info.id, draft.warehouseId, dispatchedQty],
   487→          );
   488→          if (consumed == 0) {
   489→            throw _FlowError(
   490→              'الكمية غير متوفرة للصنف «${info.name}» بالمخزن — '
   491→              'الرصيد الدفتري أقل من المطلوب (${_num(dispatchedQty)}'
   492→              '${freeQty > _qtyEpsilon ? ' منها ${_num(freeQty)} مجاني' : ''}) — '
   493→              'لا يسمح النظام بمخزون سالب.',
   494→            );
   495→          }
   496→        }
   497→
   498→        // 8-د) تجميع التكلفة على رأس الفاتورة (COGS — ملحق و).
   499→        await txn.update(
   500→          'invoice',
   501→          {'cost_total': costTotal},
   502→          where: 'id = ?',
   503→          whereArgs: [invoiceId],
   504→        );
   505→
   506→        // 8-هـ) الجزء النقدي: سند قبض بالصندوق الافتراضي بعملة الفاتورة
   507→        //       وسعرها + تخصيصه للفاتورة (القرار 2 برأس الملف).
   508→        if (netPaid > moneyEpsilon) {
   509→          final cashTxId = await txn.insert('cash_tx', {
   510→            'tx_type': 'receipt',
   511→            'cashbox_id': cashboxId,
   512→            'currency_id': draft.currencyId,
   513→            'amount': netPaid, // الصافي فقط — الباقي خارج الصندوق.
   514→            'exchange_rate': exchangeRate,
   515→            'tx_date': issuedIso,
   516→            'ref_type': 'invoice',
   517→            'ref_id': invoiceId,
   518→            'customer_id': null, // لا يخصم من رصيد العميل (القرار 2).
   519→            'description': 'تحصيل نقدي عند إصدار $invoiceNo',
   520→            'created_at': at.toUtc().toIso8601String(),
   521→            'created_by': userId,
   522→          });
   523→          await txn.insert('payment_allocation', {
   524→            'cash_tx_id': cashTxId,
   525→            'invoice_id': invoiceId,
   526→            'allocated_amount': netPaid,
   527→            'allocated_at': at.toUtc().toIso8601String(),
   528→            'created_by': userId,
   529→          });
   530→        }
   531→        // الجزء الآجل لا يُدرج في payment_allocation (لا سند له) — دينه
   532→        // في invoice.due_amount تقرأه صيغة رصيد العميل (FR-03-02).
   533→
   534→        // 8-و) تحويل عرض السعر — ذرّياً داخل نفس المعاملة (FR-02-11).
   535→        if (draft.fromQuotationId != null) {
   536→          final converted = await txn.update(
   537→            'quotation',
   538→            {
   539→              'status': 'converted',
   540→              'converted_invoice_id': invoiceId,
   541→              'updated_at': at.toUtc().toIso8601String(),
   542→            },
   543→            where: "id = ? AND status IN ('draft','sent')",
   544→            whereArgs: [draft.fromQuotationId],
   545→          );
   546→          if (converted == 0) {
   547→            throw _FlowError(
   548→              'عرض السعر غير قابل للتحويل — حُوِّل أو أُلغي للتو من جهة أخرى.',
   549→            );
   550→          }
   551→        }
   552→
   553→        // 8-ز) قيد التدقيق (append-only — نمط المستودعات القائمة).
   554→        await txn.insert('audit_log', {
   555→          'user_id': userId,
   556→          'action': 'sale_post',
   557→          'entity': 'invoice',
   558→          'entity_id': invoiceId,
   559→          'details':
   560→              'no=$invoiceNo total=${totals.grandTotal} '
   561→              'paid=$netPaid due=$remainingCredit '
   562→              'currency=$currencyCode rate=$exchangeRate'
   563→              '${rateIsFallback ? ' fallback=1' : ''}'
   564→              '${freeTotal > _qtyEpsilon ? ' free=${_num(freeTotal)}' : ''}',
   565→          'at': at.toUtc().toIso8601String(),
   566→        });
   567→
   568→        return Ok<SalePostedReceipt, String>(
   569→          SalePostedReceipt(
   570→            invoiceId: invoiceId,
   571→            invoiceNo: invoiceNo,
   572→            totals: totals,
   573→            payStatus: payStatus,
   574→            exchangeRate: exchangeRate,
   575→            rateIsFallback: rateIsFallback,
   576→            changeDue: changeDue,
   577→            remainingCredit: remainingCredit,
   578→          ),
   579→        );
   580→      });
   581→    } on _FlowError catch (e) {
   582→      return Err(e.message);
   583→    } on StateError catch (e) {
   584→      // حراس المستودعات الشريكة (applyAllocation وغيره) — رسالتهم عربية
   585→      // جاهزة وتدحرج المعاملة قبل الوصول هنا.
   586→      return Err(e.message);
   587→    } on DatabaseException catch (e) {
   588→      return Err(_describeDbError(e));
   589→    }
   590→  }

   753→  /// فحص توفر المخزون المسبق (قاعدة 5.4-5): يجمع المطلوب لكل صنف غير
   754→  /// خدمي عبر الأسطر (الأسطر المكررة تُجمع) ويقارنه بالمتاح:
   755→  /// العادي من `stock_level`، والمتتبع من دفعات FEFO النشطة غير المنتهية.
   756→  ///
   757→  /// البونص (UX-4): المجموع **بالمنصرف الكلي** qty + freeQty — المجاني
   758→  /// وحدات تخرج من المخزون مثل المدفوع تماماً.
   759→  Future<String?> _checkAvailability(
   760→    List<CartLine> lines,
   761→    Map<int, _ProductInfo> products,
   762→    int warehouseId,
   763→    DateTime asOf,
   764→  ) async {
   765→    final requested = <int, double>{};
   766→    final requestedFree = <int, double>{};
   767→    for (final line in lines) {
   768→      final info = products[line.productId]!;
   769→      if (info.isService) continue;
   770→      requested[info.id] =
   771→          (requested[info.id] ?? 0) + line.qty + line.freeQty;
   772→      if (line.freeQty > _qtyEpsilon) {
   773→        requestedFree[info.id] = (requestedFree[info.id] ?? 0) + line.freeQty;
   774→      }
   775→    }
   776→    if (requested.isEmpty) return null;
   777→
   778→    final cutoff = _dateOnly(asOf);
   779→    final ids = requested.keys.toList();
   780→    final inClause = List.filled(ids.length, '?').join(',');
   781→
   782→    // العادي: دفتر stock_level هو الحقيقة.
   783→    final plainRows = await _db.rawQuery(
   784→      'SELECT product_id, qty FROM stock_level '
   785→      'WHERE warehouse_id = ? AND product_id IN ($inClause)',
   786→      [warehouseId, ...ids],
   787→    );
   788→    final plainQty = {
   789→      for (final row in plainRows)
   790→        row['product_id'] as int: (row['qty'] as num?)?.toDouble() ?? 0,
   791→    };
   792→
   793→    // المتتبع: مجموع الدفعات النشطة غير المنتهية (نفس شرط FEFO).
   794→    final batchRows = await _db.rawQuery(
   795→      'SELECT product_id, SUM(qty) AS q FROM batch '
   796→      'WHERE warehouse_id = ? AND product_id IN ($inClause) '
   797→      '  AND qty > 0 AND is_archived = 0 AND expiry_date >= ? '
   798→      'GROUP BY product_id',
   799→      [warehouseId, ...ids, cutoff],
   800→    );
   801→    final batchQty = {
   802→      for (final row in batchRows)
   803→        row['product_id'] as int: (row['q'] as num?)?.toDouble() ?? 0,
   804→    };
   805→
   806→    for (final entry in requested.entries) {
   807→      final info = products[entry.key]!;
   808→      final available = info.trackBatches
   809→          ? (batchQty[entry.key] ?? 0)
   810→          : (plainQty[entry.key] ?? 0);
   811→      if (available + _qtyEpsilon < entry.value) {
   812→        final free = requestedFree[entry.key] ?? 0;
   813→        return 'الكمية غير متوفرة للصنف «${info.name}»: المتاح '
   814→            '${_num(available)} والمطلوب ${_num(entry.value)}'
   815→            '${free > _qtyEpsilon ? ' (منها ${_num(free)} مجاني)' : ''} — '
   816→            'لا يسمح النظام بمخزون سالب.';
   817→      }
   818→    }
   819→    return null;
   820→  }

   822→  /// يصوغ خطأ قاعدة البيانات بكلمات المستخدم (نمط item_repository).
   823→  String _describeDbError(DatabaseException e) {
   824→    if (e.isUniqueConstraintError()) {
   825→      final text = e.toString();
   826→      if (text.contains('invoice_no')) {
   827→        return 'تعارض في رقم الفاتورة — أعد الحفظ';
   828→      }
   829→      if (text.contains('payment_allocation')) {
   830→        return 'تخصيص مدفوعات مكرر لنفس السند والفاتورة';
   831→      }
   832→      return 'قيمة مكررة تخالف قيد التفرد في القاعدة';
   833→    }
   834→    if (_isCheckFailure(e)) {
   835→      final text = e.toString();
   836→      if (text.contains('stock_level')) {
   837→        return 'المخزون لا يسمح بخصم بهذه الكمية (رصيد سالب ممنوع)';
   838→      }
   839→      if (text.contains('due_amount') || text.contains('paid_amount')) {
   840→        return 'مبالغ الدفع غير متسقة مع إجمالي الفاتورة';
   841→      }
   842→      return 'قيمة تخالف قيد سلامة محاسبي في القاعدة';
   843→    }
   844→    return 'تعذر حفظ الفاتورة في القاعدة: $e';
   845→  }
