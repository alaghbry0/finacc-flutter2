    86→  /// ردود الفعل للحالات العابرة: بوابة FX (FR-08-09) وإشعارات الباركود.
    87→  void _onCartChanged() {
    88→    final vm = context.read<SellCartViewModel>();
    89→    final state = vm.state;
    90→    if (state.fxGateRequired && !_fxGateOpen) {
    91→      final currency = state.selectedCurrency;
    92→      if (currency != null && !currency.isBase) {
    93→        _fxGateOpen = true;
    94→        WidgetsBinding.instance.addPostFrameCallback((_) async {
    95→          final saved = await showFxRateGateSheet(
    96→            context,
    97→            cartVm: vm,
    98→            currencyCode: currency.code,
    99→          );
   100→          _fxGateOpen = false;
   101→          vm.clearFxGate();
   102→          if (!saved || !mounted) return;
   103→          final l10n = AppLocalizations.of(context)!;
   104→          ScaffoldMessenger.of(context)
   105→            ..hideCurrentSnackBar()
   106→            ..showSnackBar(SnackBar(content: Text(l10n.sellFxSavedAndResumed)));
   107→        });
   108→      } else {
   109→        vm.clearFxGate();
   110→      }
   111→    }
   112→    if (state.notice != null && !_noticeShown) {
   113→      _noticeShown = true;
   114→      final message = state.notice!;
   115→      WidgetsBinding.instance.addPostFrameCallback((_) {
   116→        vm.clearNotice();
   117→        _noticeShown = false;
   118→        if (!mounted) return;
   119→        ScaffoldMessenger.of(context)
   120→          ..hideCurrentSnackBar()
   121→          ..showSnackBar(SnackBar(content: Text(message)));
   122→      });
   123→    }
   124→  }

   154→  Future<void> _openPayment() async {
   155→    final app = context.read<AppController>();
   156→    final vm = context.read<SellCartViewModel>();
   157→    final state = vm.state;
   158→    final priced = vm.pricedCart;
   159→    if (priced == null) return;
   160→    final posted = await showPaymentSheet(
   161→      context,
   162→      grandTotal: priced.totals.grandTotal,
   163→      decimals: state.selectedCurrency?.decimals ?? 2,
   164→      currencyCode: state.selectedCurrency?.code,
   165→      customerName: state.customer?.name,
   166→      onConfirm: (paidCash, method) =>
   167→          vm.postSale(paidCash: paidCash, method: method),
   168→      // FR-03-05 (17-c): بوابة حد الائتمان — تُجلب لحظة التأكيد من
   169→      // المستودع (الحد + الرصيد الحي بعملة الفاتورة) والإعداد.
   170→      creditGate: () => _resolveCreditGate(app, vm),
   171→    );
   172→    if (posted) {
   173→      vm.dismissReceipt();
   174→    }
   175→  }

   177→  /// يجلب معطيات بوابة حد الائتمان — null عند غياب العميل/المستودعات
   178→  /// (عميل نقدي مجهول: لا آجل أصلاً فلا فحص — FR-03-05).
   179→  Future<CreditLimitGate?> _resolveCreditGate(
   180→    AppController app,
   181→    SellCartViewModel vm,
   182→  ) async {
   183→    final customerId = vm.state.customer?.id;
   184→    final currencyId = vm.state.currencyId;
   185→    if (customerId == null || currencyId == null) return null;
   186→    final customers = app.customers;
   187→    final settings = app.settings;
   188→    if (customers == null || settings == null) return null;
   189→    // checkCredit يقرأ الحد من جدول العميل والرصيد بصيغة FR-03-02
   190→    // بعملة الفاتورة (لا الحقل المخزَّن في السلة — دائماً حي).
   191→    final check = await customers.checkCredit(customerId, currencyId, 0);
   192→    final action = await settings.getString(
   193→      'parties.credit_limit_action',
   194→      'warn',
   195→    );
   196→    return CreditLimitGate(
   197→      creditLimit: check.creditLimit,
   198→      action: action,
   199→      currentBalance: check.balance,
   200→    );
   201→  }
