   243→  Future<void> load() async {
   244→    if (_initialized) {
   245→      // إعادة الدخول للشاشة: تحديث سعر اليوم فقط (السلة كما هي).
   246→      await _refreshRateKnowledge();
   247→      return;
   248→    }
   249→    _state = SellCartState.initial;
   250→    notifyListeners();
   251→    try {
   252→      final results = await Future.wait<Object?>([
   253→        _companies.listActiveCurrencies(),
   254→        _companies.findBaseCurrency(),
   255→        _companies.findDefaultWarehouseId(),
   256→        _companies.findAdminUserId(),
   257→      ]);
   258→      final currencies = results[0]! as List<Currency>;
   259→      final base = results[1] as Currency?;
   260→      final warehouseId = results[2] as int?;
   261→      _userId = results[3] as int?;
   262→
   263→      _initialized = true;
   264→      _state = SellCartState(
   265→        loading: false,
   266→        currencies: currencies,
   267→        baseCurrency: base,
   268→        currencyId: base?.id,
   269→        todayRate: null,
   270→        rateKnown: true,
   271→        customer: null,
   272→        warehouseId: warehouseId,
   273→        lines: const <CartUiLine>[],
   274→        invoiceDiscountType: SaleDiscountType.amount,
   275→        invoiceDiscountValue: 0,
   276→        posting: false,
   277→        nextInvoiceNo: await _nextInvoicePreview(),
   278→      );
   279→    } catch (error) {
   280→      _state = SellCartState(
   281→        loading: false,
   282→        error: error,
   283→        currencies: const <Currency>[],
   284→        rateKnown: true,
   285→        lines: const <CartUiLine>[],
   286→        invoiceDiscountType: SaleDiscountType.amount,
   287→        invoiceDiscountValue: 0,
   288→        posting: false,
   289→      );
   290→    }
   291→    notifyListeners();
   292→  }

   357→  /// اختيار عملة الفاتورة — عند عملة غير الأساس بلا سعر اليوم تُفتح
   358→  /// بوابة الإدخال فوراً (FR-08-09).
   359→  Future<void> setCurrency(int currencyId) async {
   360→    if (currencyId == _state.currencyId) return;
   361→    final currency = _currencyById(currencyId);
   362→    final isBase = currency?.isBase ?? false;
   363→    var rateKnown = true;
   364→    double? rate;
   365→    if (!isBase) {
   366→      rate = await _fx.rateFor(currencyId, _clock());
   367→      rateKnown = rate != null && rate > 0;
   368→    }
   369→    _state = _stateWith(
   370→      currencyId: currencyId,
   371→      todayRate: rate,
   372→      rateKnown: rateKnown,
   373→      fxGateRequired: !isBase && !rateKnown,
   374→      postError: null,
   375→      notice: null,
   376→    );
   377→    notifyListeners();
   378→  }

   380→  /// حفظ سعر اليوم من بوابة FX (يكتب عبر fx.setRate ثم يتابع).
   381→  Future<Result<void, String>> saveTodayRate(double rate) async {
   382→    final currencyId = _state.currencyId;
   383→    if (currencyId == null) {
   384→      return const Err<void, String>('اختر عملة أولاً.');
   385→    }
   386→    final result = await _fx.setRate(
   387→      currencyId: currencyId,
   388→      date: _clock(),
   389→      rate: rate,
   390→      userId: _userId ?? 1,
   391→      now: _clock(),
   392→    );
   393→    if (result.isOk) {
   394→      await _refreshRateKnowledge();
   395→    }
   396→    return result;
   397→  }

   399→  /// استهلاك طلب فتح بوابة FX (بعد فتح النافذة).
   400→  void clearFxGate() {
   401→    if (!_state.fxGateRequired) return;
   402→    _state = _stateWith(fxGateRequired: false);
   403→    notifyListeners();
   404→  }

   406→  Future<void> _refreshRateKnowledge() async {
   407→    final currencyId = _state.currencyId;
   408→    if (currencyId == null) return;
   409→    final currency = _currencyById(currencyId);
   410→    if (currency == null || currency.isBase) {
   411→      _state = _stateWith(
   412→        todayRate: null,
   413→        rateKnown: true,
   414→        fxGateRequired: false,
   415→      );
   416→      notifyListeners();
   417→      return;
   418→    }
   419→    final rate = await _fx.rateFor(currencyId, _clock());
   420→    final known = rate != null && rate > 0;
   421→    _state = _stateWith(
   422→      todayRate: rate,
   423→      rateKnown: known,
   424→      fxGateRequired: known ? false : _state.fxGateRequired,
   425→    );
   426→    notifyListeners();
   427→  }

   429→  Currency? _currencyById(int id) {
   430→    for (final currency in _state.currencies) {
   431→      if (currency.id == id) return currency;
   432→    }
   433→    return null;
   434→  }

   440→  /// إضافة صنف (الكمية 1 أو دمج مع سطر موجود لنفس الصنف).
   441→  void addItemFromInfo(ItemStockInfo info) {
   442→    final warehouseId = _state.warehouseId;
   443→    final available = info.item.isService
   444→        ? null
   445→        : (info.warehouseQty[warehouseId] ?? info.totalQty);
   446→    final price = info.retailPrice ?? 0.0;
   447→    final existing = _lineIndexOf(info.item.id);
   448→    if (existing >= 0) {
   449→      _updateLine(existing, qty: _state.lines[existing].qty + 1);
   450→      return;
   451→    }
   452→    _state = _stateWith(
   453→      lines: [
   454→        ..._state.lines,
   455→        CartUiLine(
   456→          productId: info.item.id,
   457→          name: info.item.name,
   458→          barcode: info.item.barcode,
   459→          qty: 1,
   460→          unitPrice: price,
   461→          discountType: SaleDiscountType.amount,
   462→          discountValue: 0,
   463→          availableQty: available,
   464→          isService: info.item.isService,
   465→        ),
   466→      ],
   467→      postError: null,
   468→    );
   469→    notifyListeners();
   470→  }

   472→  /// مسح باركود بإدخال يدوي — أول تطابق يضاف مباشرة بكمية 1.
   473→  Future<bool> addByBarcode(String code) async {
   474→    final trimmed = code.trim();
   475→    if (trimmed.isEmpty) return false;
   476→    final info = await _items.findByBarcode(
   477→      trimmed,
   478→      currencyId: _state.currencyId,
   479→    );
   480→    if (info == null) {
   481→      _notice('لا صنف بباركود «$trimmed» — تأكد من الرقم أو أضفه من القائمة.');
   482→      return false;
   483→    }
   484→    addItemFromInfo(info);
   485→    return true;
   486→  }

   488→  /// تعديل الكمية — القيم غير الصالحة (≤ 0 أو غير رقمية) ترفض فوراً.

   491→      _notice('الكمية يجب أن تكون رقماً أكبر من صفر.');

   501→      return;
   502→    }

   506→  /// تعديل خصم السطر (٪ ضمن 0–100 / مبلغ ≥ 0).
   507→  void setLineDiscount(int index, SaleDiscountType type, double value) {
   508→    if (value.isNaN || value.isInfinite || value < 0) {
   509→      _notice('قيمة الخصم لا يمكن أن تكون سالبة.');
   510→      return;
   511→    }
   512→    if (type == SaleDiscountType.percent && value > 100) {
   513→      _notice('نسبة الخصم لا يمكن أن تتجاوز 100%.');
   514→      return;
   515→    }
   516→    _updateLine(index, discountType: type, discountValue: value);
   517→  }

   519→  /// حذف سطر بالفهرس.

   523→    _state = _stateWith(lines: lines);
   524→    notifyListeners();
   525→  }

   538→    notifyListeners();

   545→  /// بناء مسودة الترحيل من الحالة الحالية (للاستخدام الداخلي والاختبارات).
   546→  SaleDraft? buildDraft({
   547→    required double paidCash,
   548→    required SalePaymentMethod method,
   549→  }) {
   550→    final currencyId = _state.currencyId;
   551→    final warehouseId = _state.warehouseId;
   552→    if (currencyId == null || warehouseId == null) return null;
   553→    return SaleDraft(
   554→      customerId: _state.customer?.id,
   555→      currencyId: currencyId,
   556→      lines: cartLines,
   557→      invoiceDiscountType: _state.invoiceDiscountType,
   558→      invoiceDiscountValue: _state.invoiceDiscountValue,
   559→      paidCash: paidCash,
   560→      paymentMethod: method,
   561→      warehouseId: warehouseId,
   562→      issuedAt: _clock(),
   563→    );
   564→  }

   566→  /// **ترحيل الفاتورة** — نجاح: تُفرَّغ السلة ويُحفظ الإيصال؛ فشل: تُعرض
   567→  /// رسالة الرفض العربية **والسلة لا تُفقد**.
   568→  Future<Result<SalePostedReceipt, String>> postSale({
   569→    required double paidCash,
   570→    required SalePaymentMethod method,
   571→  }) async {
   572→    if (_state.posting) {
   573→      return const Err<SalePostedReceipt, String>(
   574→        'ترحيل جارٍ بالفعل — لحظات ويكتمل.',
   575→      );
   576→    }
   577→    final failure = _prePostFailure(paidCash, method);
   578→    if (failure != null) {
   579→      _state = _stateWith(postError: failure);
   580→      notifyListeners();
   581→      return Err<SalePostedReceipt, String>(failure);
   582→    }
   583→    final draft = buildDraft(paidCash: paidCash, method: method);
   584→    if (draft == null) {
   585→      const message = 'الإعدادات غير مكتملة (العملة/المخزن) — أعد فتح الشاشة.';
   586→      _state = _stateWith(postError: message);
   587→      notifyListeners();
   588→      return const Err<SalePostedReceipt, String>(message);
   589→    }
   590→
   591→    _state = _stateWith(posting: true, postError: null);
   592→    notifyListeners();
   593→    final result = await _sales.postSale(
   594→      draft,
   595→      userId: _userId ?? 1,
   596→      now: _clock(),
   597→    );
   598→    if (result.isOk) {
   599→      final receipt = result.valueOrNull!;
   600→      _state = _stateWith(
   601→        posting: false,
   602→        lines: const <CartUiLine>[],
   603→        invoiceDiscountType: SaleDiscountType.amount,
   604→        invoiceDiscountValue: 0,
   605→        lastReceipt: receipt,
   606→        postError: null,
   607→        notice: null,
   608→        nextInvoiceNo: await _nextInvoicePreview(),
   609→      );
   610→      notifyListeners();
   611→    } else {
   612→      _state = _stateWith(posting: false, postError: result.errorOrNull!);
   613→      notifyListeners();
   614→    }
   615→    return result;
   616→  }

   618→  /// فحوص ما قبل الترحيل (بترتيب رسائل واضح للمستخدم) — null عند السلامة.
   619→  String? _prePostFailure(double paidCash, SalePaymentMethod method) {
   620→    if (_state.lines.isEmpty) {
   621→      return 'أضف بنداً واحداً على الأقل إلى الفاتورة قبل الدفع.';
   622→    }
   623→    final cartFailure = cartError;
   624→    if (cartFailure != null) return cartFailure;
   625→    final currency = _state.selectedCurrency;
   626→    if (currency == null) return 'اختر عملة الفاتورة أولاً.';
   627→    if (!currency.isBase && !_state.rateKnown) {
   628→      // FR-08-09: منع الحفظ + طلب فتح بوابة إدخال سعر اليوم.
   629→      _state = _stateWith(fxGateRequired: true);
   630→      notifyListeners();
   631→      return 'لا يوجد سعر صرف لعملة ${currency.code} بتاريخ اليوم — '
   632→          'أدخل سعر اليوم أولاً ثم أكمل الدفع.';
   633→    }
   634→    final total = grandTotal;
   635→    final payFailure = SalePricing.validatePayment(total, paidCash, method);
   636→    if (payFailure != null) return payFailure;
   637→    // عميل نقدي مجهول: نقدي كامل حصراً (الآجل يحتاج حساباً معروفاً).
   638→    final settlement = SalePricing.settlePayment(total, paidCash);
   639→    if (settlement.remainingCredit > moneyEpsilon &&
   640→        !_state.isCustomerSelected) {
   641→      return 'البيع الآجل يتطلب اختيار عميل أولاً — العميل النقدي المجهول '
   642→          'يسدد نقداً كاملاً فقط.';
   643→    }
   644→    return null;
   645→  }

   648→  // عرض السعر (FR-02-11)
   649→  // ───────────────────────────────────────────────────────────────────
   650→
   651→  /// مدة الصلاحية الافتراضية لعرض السعر المحفوظ من السلة (P2-5) —
   652→  /// +30 يوماً إن لم يمرّر المستدعي صلاحية صريحة.
   653→  static const Duration quotationDefaultValidity = Duration(days: 30);
   654→
   655→  /// حفظ السلة الحالية كعرض سعر QTE — بلا أي حركة مخزون/صندوق.
   656→  /// السلة تبقى كما هي (يمكن المتابعة في البيع أو التفريغ من الشاشة).
   657→  ///
   658→  /// [validUntil] (P2-5): صلاحية صريحة — والافتراضي +30 يوماً من لحظة
   659→  /// الحفظ (مسار الحفظ لا يمرّ بنموذج إدخال في هذه الجولة).
   660→  Future<Result<Quotation, String>> saveAsQuotation({
   661→    DateTime? validUntil,
   662→  }) async {
   663→    if (_state.lines.isEmpty) {
   664→      return const Err<Quotation, String>(
   665→        'أضف بنداً واحداً على الأقل قبل حفظ عرض السعر.',
   666→      );
   667→    }
   668→    final cartFailure = cartError;
   669→    if (cartFailure != null) {
   670→      return Err<Quotation, String>(cartFailure);
   671→    }
   672→    final currencyId = _state.currencyId;
   673→    final warehouseId = _state.warehouseId;
   674→    if (currencyId == null || warehouseId == null) {
   675→      return const Err<Quotation, String>(
   676→        'الإعدادات غير مكتملة (العملة/المخزن) — أعد فتح الشاشة.',
   677→      );
   678→    }
   679→    return _quotations.createQuotation(
   680→      QuotationDraft(
   681→        customerId: _state.customer?.id,
   682→        currencyId: currencyId,
   683→        warehouseId: warehouseId,
   684→        lines: [
   685→          for (final line in _state.lines)
   686→            QuotationLine(
   687→              productId: line.productId,
   688→              qty: line.qty,
   689→              unitPrice: line.unitPrice,
   690→              lineDiscountType: line.discountType,
   691→              lineDiscountValue: line.discountValue,
   692→            ),
   693→        ],
   694→        invoiceDiscountType: _state.invoiceDiscountType,
   695→        invoiceDiscountValue: _state.invoiceDiscountValue,
   696→        // P2-5: الصلاحية الافتراضية +30 يوماً عند غياب صلاحية صريحة.
   697→        validUntil: validUntil ?? _clock().add(quotationDefaultValidity),
   698→        issuedAt: _clock(),
   699→      ),
   700→      userId: _userId ?? 1,
   701→      now: _clock(),
   702→    );
   703→  }

   781→  /// نسخة حالة محدَّثة — للحقول القابلة للتصفير (customer/postError/notice/
   782→  /// todayRate/lastReceipt): حذف الوسيطة أو تمرير null يصفّرها، وتمرير
   783→  /// `_keep` يبقيها. بقية الحقول: حذف الوسيطة يبقيها.
   784→  SellCartState _stateWith({
   785→    bool? loading,
   786→    List<Currency>? currencies,
   787→    Object? baseCurrency = _keep,
   788→    int? currencyId,
   789→    Object? todayRate = _keep,
   790→    bool? rateKnown,
   791→    Object? customer = _keep,
   792→    int? warehouseId,
   793→    List<CartUiLine>? lines,
   794→    SaleDiscountType? invoiceDiscountType,
   795→    double? invoiceDiscountValue,
   796→    bool? posting,
   797→    Object? postError = _keep,
   798→    Object? notice = _keep,
   799→    String? nextInvoiceNo,
   800→    bool? fxGateRequired,
   801→    Object? lastReceipt = _keep,
   802→  }) => SellCartState(
   803→    loading: loading ?? _state.loading,
   804→    error: _state.error,
   805→    currencies: currencies ?? _state.currencies,
   806→    baseCurrency: identical(baseCurrency, _keep)
   807→        ? _state.baseCurrency
   808→        : baseCurrency as Currency?,
   809→    currencyId: currencyId ?? _state.currencyId,
   810→    todayRate: identical(todayRate, _keep)
   811→        ? _state.todayRate
   812→        : todayRate as double?,
   813→    rateKnown: rateKnown ?? _state.rateKnown,
   814→    customer: identical(customer, _keep)
   815→        ? _state.customer
   816→        : customer as CartCustomer?,
   817→    warehouseId: warehouseId ?? _state.warehouseId,
   818→    lines: lines ?? _state.lines,
   819→    invoiceDiscountType: invoiceDiscountType ?? _state.invoiceDiscountType,
   820→    invoiceDiscountValue: invoiceDiscountValue ?? _state.invoiceDiscountValue,
   821→    posting: posting ?? _state.posting,
   822→    postError: identical(postError, _keep)
   823→        ? _state.postError
   824→        : postError as String?,
   825→    notice: identical(notice, _keep) ? _state.notice : notice as String?,
   826→    nextInvoiceNo: nextInvoiceNo ?? _state.nextInvoiceNo,
   827→    fxGateRequired: fxGateRequired ?? _state.fxGateRequired,
   828→    lastReceipt: identical(lastReceipt, _keep)
   829→        ? _state.lastReceipt
   830→        : lastReceipt as SalePostedReceipt?,
   831→  );
