    29→class MovementsScreen extends StatefulWidget {

    38→class _MovementsScreenState extends State<MovementsScreen> {
    39→  MovementsViewModel? _vm;
    40→  List<CashboxWithBalance> _boxes = const <CashboxWithBalance>[];
    41→  Object? _boxesError;
    42→
    43→  @override
    44→  void initState() {
    45→    super.initState();
    46→    final app = context.read<AppController>();
    47→    final vm = MovementsViewModel(
    48→      cashRepo: app.cash!,
    49→      companyRepo: app.companies!,
    50→    );
    51→    _vm = vm;
    52→    if (widget.initialBoxId != null) {
    53→      vm.setBoxFilter(widget.initialBoxId);
    54→    }
    55→    unawaited(vm.load());
    56→    // قائمة الصناديق لمرشح الصندوق — تحميل مساعد في الشاشة (النموذج
    57→    // يكشف setBoxFilter فقط) عبر مستودع AppController؛ موثق في سجل
    58→    // العمل 12.
    59→    unawaited(_loadBoxes(app));
    60→  }
    61→
    62→  Future<void> _loadBoxes(AppController app) async {
    63→    try {
    64→      final boxes = await app.cash!.listBoxes();
    65→      if (!mounted) return;
    66→      setState(() {
    67→        _boxes = boxes;
    68→        _boxesError = null;
    69→      });
    70→    } catch (error) {
    71→      if (!mounted) return;
    72→      setState(() => _boxesError = error);
    73→    }
    74→  }
    75→
    76→  @override
    77→  Widget build(BuildContext context) {
    78→    final vm = _vm;
    79→    if (vm == null) {
    80→      return const Scaffold(body: Center(child: CircularProgressIndicator()));
    81→    }
    82→    return ChangeNotifierProvider<MovementsViewModel>.value(
    83→      value: vm,
    84→      // تحديث حي عند تبديل التبويب/الرجوع (درس §10) — المرشحات محفوظة
    85→      // في النموذج فتُطبَّق عند إعادة التحميل.
    86→      child: RefreshOnActive(
    87→        routePattern: RegExp(r'^/cash/movements$'),
    88→        onActivate: vm.load,
    89→        child: _MovementsBody(boxes: _boxes, boxesError: _boxesError),
    90→      ),
    91→    );
    92→  }
    93→}

   277→  Widget _buildContent(BuildContext context, MovementsViewModel vm) {
   278→    final l10n = AppLocalizations.of(context)!;
   279→    final state = vm.state;
   280→    if (state.loading) {
   281→      return ListView(
   282→        padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   283→        children: const [ListSkeleton(rows: 7)],
   284→      );
   285→    }
   286→    if (state.error != null) {
   287→      return ListView(
   288→        padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   289→        children: [
   290→          ErrorState(
   291→            title: l10n.genericErrorTitle,
   292→            message: l10n.dbOpenErrorMessage,
   293→            technicalDetails: state.error.toString(),
   294→            retryLabel: l10n.commonRetry,
   295→            onRetry: vm.load,
   296→            compact: true,
   297→          ),
   298→        ],
   299→      );
   300→    }
   301→    final filtered =
   302→        vm.search.trim().isNotEmpty ||
   303→        vm.boxFilter != null ||
   304→        vm.typeFilter != null ||
   305→        vm.period != CashPeriodFilter.all;
   306→    if (state.movements.isEmpty) {
   307→      return ListView(
   308→        padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   309→        children: [
   310→          EmptyState(
   311→            icon: Icons.receipt_long_rounded,
   312→            title: filtered
   313→                ? l10n.cashMovementsNoResultsTitle
   314→                : l10n.cashMovementsEmptyTitle,
   315→            message: filtered
   316→                ? l10n.cashMovementsNoResultsBody
   317→                : l10n.cashMovementsEmptyBody,
   318→            compact: true,
   319→          ),
   320→        ],
   321→      );
   322→    }
   323→    return ListView.builder(
   324→      padding: const EdgeInsets.fromLTRB(20, 4, 20, 32),
   325→      itemCount: state.movements.length,
   326→      itemBuilder: (context, index) {
   327→        final movement = state.movements[index];
   328→        return Padding(
   329→          padding: const EdgeInsets.only(bottom: 10),
   330→          child: FinCard(
   331→            padding: const EdgeInsets.all(12),
   332→            onTap: () => unawaited(
   333→              showMovementDetailSheet(context, vm: vm, movementId: movement.id),
   334→            ),
   335→            child: CashMovementTile(
   336→              movement: movement,
   337→              dimmed: movement.isVoided,
   338→            ),
   339→          ),
   340→        );
   341→      },
   342→    );
   343→  }

   407→    final l10n = AppLocalizations.of(context)!;
   408→    final reasonController = TextEditingController();
   409→    final confirmed = await showDialog<bool>(
   410→      context: context,
   411→      builder: (dialogContext) => AlertDialog(
   412→        title: Text(l10n.cashMovementVoidTitle),

   465→          ..hideCurrentSnackBar()

   467→            SnackBar(
   468→              backgroundColor: Theme.of(context).colorScheme.errorContainer,

   472→                  color: Theme.of(context).colorScheme.onErrorContainer,
   473→                ),
   474→              ),
   475→            ),
