   339→  static String _hijriText(bool arabicIndic) {
   340→    final hijri = HijriCalendar.today().toString();
   341→    // الهجري يُخرَج بأرقام غربية — يُحوَّل للشرقية عند تفعيل النظام العربي.
   342→    return arabicIndic ? Numerals.toArabicIndic(hijri) : hijri;
   343→  }
   344→
   345→  static String _gregorianText(bool arabicIndic) {
   346→    // الميلادي بـ intl العربي يخرج بأرقام شرقية — يُوحَّد مع النظام المختار.
   347→    final text = DateFormat('EEEE، d MMMM y', 'ar').format(DateTime.now());
   348→    return arabicIndic ? text : Numerals.toWestern(text);
   349→  }
   350→}

  1348→class _PatternPainter extends BoxPainter {
  1349→  _PatternPainter({required this.color});
  1350→
  1351→  final Color color;
  1352→
  1353→  @override
  1354→  void paint(Canvas canvas, Offset offset, ImageConfiguration configuration) {
  1355→    final size = configuration.size;
  1356→    if (size == null) return;
  1357→    final rect = offset & size;
  1358→    canvas.save();
  1359→    try {
  1360→      // قصّ داخل البطاقة حتى لا تتجاوز الزخرفة حوافها المستديرة.
  1361→      canvas.clipRRect(

  1363→      );
  1364→      final paint = Paint()
  1365→        ..style = PaintingStyle.stroke
  1366→        ..strokeWidth = 1
  1367→        ..color = color.withValues(alpha: 0.16);

  1376→        ..style = PaintingStyle.stroke
