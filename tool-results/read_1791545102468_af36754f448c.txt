  1348→  1348→  ReturnRefundMethod _deriveMethod(double refundTotal, double refundCash) {

  1500→  1500→          ? (batchQty[entry.key] ?? 0)
  1501→  1501→          : (plainQty[entry.key] ?? 0);

  1538→  1538→  Future<List<_BatchCap>> _loadBatchReturnCaps(

  1541→  1541→    required int warehouseId,

  1543→  1543→  }) async {

  1545→  1545→    final saleMoves = await txn.rawQuery(
  1546→  1546→      'SELECT notes, qty FROM stock_movement '
  1547→  1547→      "WHERE movement_type = 'sale' AND ref_type = 'invoice' AND ref_id = ? "

  1555→  1555→      final match = _batchNotePattern.firstMatch(
  1556→  1556→        (move['notes'] as String?) ?? '',
  1557→  1557→      );
  1558→  1558→      if (match == null) continue; // حركة غير دفعية (مخزون عام).

  1561→  1561→          (consumed[key] ?? 0) - ((move['qty'] as num?)?.toDouble() ?? 0);
  1562→  1562→      if (!order.contains(key)) order.add(key);
  1563→  1563→    }
  1564→  1564→    // ما أُعيد سابقاً لكل دفعة (حركات SRN مكتملة سابقة بنفس الصيغة).

  1566→  1566→      'SELECT sm.notes AS notes, sm.qty AS qty FROM stock_movement sm '
  1567→  1567→      "WHERE sm.movement_type = 'sale_return' AND sm.ref_type = 'invoice' "
  1568→  1568→      'AND sm.ref_id IN (SELECT id FROM invoice '

  1572→  1572→      [originalInvoiceId, productId, warehouseId],
  1573→  1573→    );
  1574→  1574→    final priorReturned = <String, double>{};
  1575→  1575→    for (final move in priorMoves) {
  1576→  1576→      final match = _batchNotePattern.firstMatch(
  1577→  1577→        (move['notes'] as String?) ?? '',
  1578→  1578→      );
  1579→  1579→      if (match == null) continue; // عودة سابقة بلا دفعة (مخزون عام).
  1580→  1580→      final key = '${match.group(1)!.trim()}|${match.group(2)!}';
  1581→  1581→      priorReturned[key] =
  1582→  1582→          (priorReturned[key] ?? 0) + ((move['qty'] as num?)?.toDouble() ?? 0);
  1583→  1583→    }
  1584→  1584→
  1585→  1585→    // السعة الفعالة لكل دفعة (بترتيب استهلاك FEFO الأصلي) مع مطابقة

  1587→  1587→    final caps = <_BatchCap>[];
  1588→  1588→    for (final key in order) {
  1589→  1589→      final effective = (consumed[key] ?? 0) - (priorReturned[key] ?? 0);
  1590→  1590→      if (effective <= _qtyEpsilon) continue;
  1591→  1591→      final parts = key.split('|');
  1592→  1592→      final batchRows = await txn.query(
  1593→  1593→        'batch',
  1594→  1594→        columns: ['id'],
  1595→  1595→        where:
  1596→  1596→            'product_id = ? AND warehouse_id = ? AND batch_number = ? '
  1597→  1597→            'AND expiry_date = ? AND is_archived = 0',
  1598→  1598→        whereArgs: [productId, warehouseId, parts[0], parts[1]],
  1599→  1599→        orderBy: 'id ASC',
  1600→  1600→        limit: 1,
  1601→  1601→      );
  1602→  1602→      if (batchRows.isEmpty) continue; // غير قابلة للتتبع → متبقٍ عام.
  1603→  1603→      caps.add(
  1604→  1604→        _BatchCap(
  1605→  1605→          batchId: batchRows.first['id'] as int,
  1606→  1606→          batchNumber: parts[0],
  1607→  1607→          expiryDate: parts[1],
  1608→  1608→          qty: effective,
  1609→  1609→        ),
  1610→  1610→      );
  1611→  1611→    }
  1612→  1612→    return caps;
  1613→  1613→  }
