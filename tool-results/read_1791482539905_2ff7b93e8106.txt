  1024→/// حجم القاعدة مقروءاً (ك.ب/م.ب) بأرقام النظام الحي — لبند «حول».
  1025→String _formatDbSize(BuildContext context, int bytes) {
  1026→  final l10n = AppLocalizations.of(context)!;
  1027→  final arabicIndic = NumeralsScope.of(context);
  1028→  String digits(String text) =>
  1029→      arabicIndic ? Numerals.toArabicIndic(text) : text;
  1030→  final kb = bytes / 1024;
  1031→  if (kb < 1024) {
  1032→    return l10n.backupSizeKb(digits(kb.round().toString()));
  1033→  }
  1034→  return l10n.backupSizeMb(digits((kb / 1024).toStringAsFixed(1)));
  1035→}
