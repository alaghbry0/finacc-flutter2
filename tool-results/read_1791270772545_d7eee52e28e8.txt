    31→class _OnboardingBody extends StatefulWidget {

    34→  @override
    35→  State<_OnboardingBody> createState() => _OnboardingBodyState();
    36→}
    37→
    38→class _OnboardingBodyState extends State<_OnboardingBody> {
    39→  List<Currency>? _currencies;
    40→  final _passphraseController = TextEditingController();
    41→  final _passphraseConfirmController = TextEditingController();
    42→  bool _obscured = true;
    43→
    44→  @override
    45→  void initState() {
    46→    super.initState();
    47→    final repo = context.read<AppController>().companies;
    48→    if (repo != null) {
    49→      unawaited(
    50→        repo.listActiveCurrencies().then((currencies) {
    51→          if (mounted) setState(() => _currencies = currencies);
    52→        }),
    53→      );
    54→    }
    55→  }
    56→
    57→  @override
    58→  void dispose() {
    59→    _passphraseController.dispose();
    60→    _passphraseConfirmController.dispose();
    61→    super.dispose();
    62→  }
    63→
    64→  @override
    65→  Widget build(BuildContext context) {
    66→    final vm = context.watch<OnboardingViewModel>();
    67→    final scheme = Theme.of(context).colorScheme;
    68→
    69→    return Scaffold(
    70→      backgroundColor: scheme.surface,
    71→      body: SafeArea(
    72→        child: AnimatedSwitcher(
    73→          duration: const Duration(milliseconds: 260),
    74→          child: switch (vm.step) {
    75→            OnboardingStep.welcome => _WelcomeStep(
    76→              key: const ValueKey('welcome'),
    77→              onStart: vm.startCompanyForm,
    78→            ),
    79→            OnboardingStep.company => _CompanyStep(
    80→              key: const ValueKey('company'),
    81→              currencies: _currencies,
    82→              vm: vm,
    83→              onContinue: vm.submitCompanyForm,
    84→            ),
    85→            OnboardingStep.security => _SecurityStep(
    86→              key: const ValueKey('security'),
    87→              vm: vm,
    88→              passphraseController: _passphraseController,
    89→              passphraseConfirmController: _passphraseConfirmController,
    90→              obscured: _obscured,
    91→              toggleObscured: () => setState(() => _obscured = !_obscured),
    92→              onSubmit: () => vm.submit(context.read<AppController>()),
    93→            ),
    94→            OnboardingStep.creating => _CreatingStep(
    95→              key: const ValueKey('creating'),
    96→            ),
    97→            OnboardingStep.done => _DoneStep(
    98→              key: const ValueKey('done'),
    99→              vm: vm,
   100→              onDone: () async {
   101→                // تفعيل الجلسة يقلب الطور — والموجّه ينتقل بـ/home تلقائياً.
   102→                await context.read<AppController>().completeOnboarding();
   103→                if (context.mounted) context.go('/home');
   104→              },
   105→            ),
   106→          },
   107→        ),
   108→      ),
   109→    );
   110→  }
   111→}

   418→class _SecurityStep extends StatelessWidget {
   419→  const _SecurityStep({
   420→    super.key,
   421→    required this.vm,
   422→    required this.passphraseController,
   423→    required this.passphraseConfirmController,
   424→    required this.obscured,
   425→    required this.toggleObscured,
   426→    required this.onSubmit,
   427→  });
   428→
   429→  final OnboardingViewModel vm;
   430→  final TextEditingController passphraseController;
   431→  final TextEditingController passphraseConfirmController;
   432→  final bool obscured;
   433→  final VoidCallback toggleObscured;
   434→  final Future<void> Function() onSubmit;
   435→
   436→  @override
   437→  Widget build(BuildContext context) {
   438→    final l10n = AppLocalizations.of(context)!;
   439→    return vm.passphraseMode
   440→        ? _passphraseSection(context, l10n)

   444→  Widget _pinSection(BuildContext context, AppLocalizations l10n) {
   445→    final scheme = Theme.of(context).colorScheme;
   446→    final errorText = _errorText(context, vm.error);
   447→    final pinReady = vm.pinForDots.length >= 4;
   448→    return ListView(
   449→      key: const ValueKey('pin-section'),
   450→      padding: const EdgeInsets.fromLTRB(24, 28, 24, 24),

   452→        _StepHeader(

   461→          length: vm.pinDotsLength,
   462→          filled: vm.pinForDots.length,

   465→        ),
   466→        const SizedBox(height: 10),

   469→          child: errorText == null

   486→            label: Text(
   487→              vm.confirmingPin ? l10n.commonConfirm : l10n.commonContinue,
   488→            ),
   489→          ),
   490→        ),
   491→      ],
   492→    );

   495→  Widget _passphraseSection(BuildContext context, AppLocalizations l10n) {

   497→    final errorText = _errorText(context, vm.error);
   498→    final valid = vm.passphraseFormValid;
   499→    final warningColor = scheme.brightness == Brightness.dark
   500→        ? const Color(0xFF39300F)
   501→        : const Color(0xFFFCEFD8);

   534→                          ?.copyWith(color: warningFg),

   537→                    Text(
   538→                      l10n.passphraseWarningBody,
   539→                      style: Theme.of(context).textTheme.bodySmall
   540→                          ?.copyWith(color: warningFg.withValues(alpha: 0.9)),
   541→                    ),
   542→                  ],
   543→                ),
   544→              ),
   545→            ],
   546→          ),
   547→        ),
   548→        const SizedBox(height: 20),
   549→        TextField(
   550→          controller: passphraseController,
   551→          onChanged: vm.setPassphrase,
   552→          obscureText: obscured,
   553→          decoration: InputDecoration(
   554→            labelText: l10n.passphraseLabel,
   555→            prefixIcon: const Icon(Icons.key_rounded),
   556→            suffixIcon: IconButton(
   557→              onPressed: toggleObscured,
   558→              icon: Icon(
   559→                obscured
   560→                    ? Icons.visibility_outlined
   561→                    : Icons.visibility_off_outlined,
   562→              ),
   563→            ),
   564→          ),
   565→        ),
   566→        const SizedBox(height: 14),
   567→        TextField(
   568→          controller: passphraseConfirmController,
   569→          onChanged: vm.setConfirmedPassphrase,
   570→          obscureText: obscured,
   571→          onSubmitted: (_) => valid ? onSubmit() : null,
   572→          decoration: InputDecoration(
   573→            labelText: l10n.passphraseConfirmLabel,
   574→            prefixIcon: const Icon(Icons.password_rounded),
   575→          ),
   576→        ),
   577→        const SizedBox(height: 10),
   578→        SizedBox(
   579→          height: 20,
   580→          child: errorText == null
   581→              ? null
   582→              : Text(
   583→                  errorText,
   584→                  style: Theme.of(context).textTheme.bodySmall
   585→                      ?.copyWith(color: scheme.error),
   586→                ),
   587→        ),
   588→        const SizedBox(height: 14),
   589→        if (vm.submitting)
   590→          const Center(child: CircularProgressIndicator())
   591→        else
   592→          SizedBox(
   593→            width: double.infinity,
   594→            child: FilledButton.icon(
   595→              onPressed: valid ? onSubmit : null,
   596→              icon: const Icon(Icons.shield_rounded, size: 20),
   597→              label: Text(l10n.commonContinue),

   599→          ),
   600→        TextButton(onPressed: vm.backToPin, child: Text(l10n.commonBack)),

   602→    );
   603→  }
   604→
   605→  String? _errorText(BuildContext context, OnboardingError? error) {
   606→    if (error == null) return null;
   607→    final l10n = AppLocalizations.of(context)!;
   608→    return switch (error) {
   609→      OnboardingError.pinMismatch => l10n.pinMismatch,
   610→      OnboardingError.pinInvalidLength => l10n.pinInvalidLength,
   611→      OnboardingError.passphraseMismatch => l10n.passphraseMismatch,
   612→      OnboardingError.passphraseShort => l10n.passphraseShort,
   613→      OnboardingError.passphraseRequired => l10n.passphraseShort,
   614→      OnboardingError.setupFailed => l10n.setupFailedBody,
   615→    };
   616→  }
   617→}
