   204→  /// ملخص حي لخيارات المظهر بصف البوابة (ثيم · أرقام · خط · تباين).
   205→  String _appearanceSummary(BuildContext context, AppController app) {
   206→    final l10n = AppLocalizations.of(context)!;
   207→    final theme = switch (app.themeMode) {
   208→      'light' => l10n.themeLight,
   209→      'dark' => l10n.themeDark,
   210→      _ => l10n.themeSystem,
   211→    };
   212→    final scale = switch (app.fontScale) {
   213→      'large' => l10n.settings2FontScaleLarge,
   214→      'xlarge' => l10n.settings2FontScaleXlarge,
   215→      _ => l10n.settings2FontScaleNormal,
   216→    };
   217→    return '$theme · $scale'
   218→        '${app.highContrast ? ' · ${l10n.settings2HighContrast}' : ''}';
   219→  }

   873→/// حجم القاعدة مقروءاً (ك.ب/م.ب) بأرقام النظام الحي — لبند «حول».
   874→String _formatDbSize(BuildContext context, int bytes) {
   875→  final l10n = AppLocalizations.of(context)!;
   876→  final arabicIndic = NumeralsScope.of(context);
   877→  String digits(String text) =>
   878→      arabicIndic ? Numerals.toArabicIndic(text) : text;
   879→  final kb = bytes / 1024;
   880→  if (kb < 1024) {
   881→    return l10n.backupSizeKb(digits(kb.round().toString()));
   882→  }
   883→  return l10n.backupSizeMb(digits((kb / 1024).toStringAsFixed(1)));
   884→}
