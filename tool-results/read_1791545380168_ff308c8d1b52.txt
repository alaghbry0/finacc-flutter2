    26→/// قائمة فواتير المبيعات (مسار مقترح `/sell/invoices`).
    27→class SalesInvoicesScreen extends StatelessWidget {
    28→  const SalesInvoicesScreen({super.key, this.viewModel});
    29→
    30→  /// Seam اختبار: نموذج محمّل مسبقاً.
    31→  final SalesInvoicesViewModel? viewModel;
    32→
    33→  @override
    34→  Widget build(BuildContext context) {
    35→    final app = context.read<AppController>();
    36→    late final SalesInvoicesViewModel vm;
    37→    if (viewModel != null) {
    38→      vm = viewModel!;

    41→      unawaited(vm.load());
    42→    }

    44→      value: vm,

    47→      child: RefreshOnActive(
    48→        routePattern: RegExp(r'^/sell/invoices$'),
    49→        onActivate: vm.load,
    50→        child: const _SalesInvoicesBody(),
    51→      ),
    52→    );
    53→  }

   114→  Widget _buildContent(BuildContext context, SalesInvoicesViewModel vm) {

   116→    final state = vm.state;
   117→    if (state.loading) {
   118→      return ListView(
   119→        padding: const EdgeInsets.fromLTRB(20, 8, 20, 32),
   120→        children: const [ListSkeleton(rows: 7)],
   121→      );
   122→    }
   123→    if (state.error != null) {
   124→      return ListView(
   125→        padding: const EdgeInsets.fromLTRB(20, 8, 20, 32),
   126→        children: [
   127→          ErrorState(
   128→            title: l10n.genericErrorTitle,
   129→            message: l10n.dbOpenErrorMessage,
   130→            technicalDetails: state.error.toString(),
   131→            retryLabel: l10n.commonRetry,
   132→            onRetry: vm.load,
   133→            compact: true,
   134→          ),
   135→        ],
   136→      );
   137→    }
   138→    if (state.invoices.isEmpty) {
   139→      return ListView(
   140→        padding: const EdgeInsets.fromLTRB(20, 8, 20, 32),
   141→        children: [
   142→          EmptyState(
   143→            icon: Icons.receipt_long_rounded,
   144→            title: l10n.sellInvoicesEmptyTitle,
   145→            message: l10n.sellInvoicesEmptyBody,
   146→            actionLabel: l10n.sellHomeNewInvoice,
   147→            onAction: () => context.go('/sell/new'),
   148→            compact: true,
   149→          ),
   150→        ],
   151→      );
   152→    }
   153→    final visible = state.visible;
   154→    if (visible.isEmpty) {
   155→      return ListView(
   156→        padding: const EdgeInsets.fromLTRB(20, 8, 20, 32),
   157→        children: [
   158→          EmptyState(
   159→            icon: Icons.search_off_rounded,
   160→            title: l10n.sellInvoicesNoResultsTitle,
   161→            message: l10n.sellInvoicesNoResultsBody,
   162→            compact: true,
   163→          ),
   164→        ],
   165→      );
   166→    }
   167→    return ListView(
   168→      padding: const EdgeInsets.fromLTRB(20, 8, 20, 32),
   169→      children: [
   170→        for (final invoice in visible)
   171→          Padding(
   172→            padding: const EdgeInsets.only(bottom: 12),
   173→            child: _InvoiceCard(invoice: invoice),
   174→          ),
   175→      ],
   176→    );
   177→  }
