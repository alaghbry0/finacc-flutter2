    91→  /// ردود الفعل للحالات العابرة: بوابة FX (FR-08-09) وإشعارات الباركود.
    92→  void _onCartChanged() {
    93→    final vm = context.read<SellCartViewModel>();
    94→    final state = vm.state;
    95→    if (state.fxGateRequired && !_fxGateOpen) {
    96→      final currency = state.selectedCurrency;
    97→      if (currency != null && !currency.isBase) {
    98→        _fxGateOpen = true;
    99→        WidgetsBinding.instance.addPostFrameCallback((_) async {
   100→          final saved = await showFxRateGateSheet(
   101→            context,
   102→            cartVm: vm,
   103→            currencyCode: currency.code,
   104→          );
   105→          _fxGateOpen = false;
   106→          vm.clearFxGate();
   107→          if (!saved || !mounted) return;
   108→          final l10n = AppLocalizations.of(context)!;
   109→          ScaffoldMessenger.of(context)
   110→            ..hideCurrentSnackBar()
   111→            ..showSnackBar(SnackBar(content: Text(l10n.sellFxSavedAndResumed)));
   112→        });
   113→      } else {
   114→        vm.clearFxGate();
   115→      }
   116→    }
   117→    if (state.notice != null && !_noticeShown) {
   118→      _noticeShown = true;
   119→      final message = state.notice!;
   120→      WidgetsBinding.instance.addPostFrameCallback((_) {
   121→        vm.clearNotice();
   122→        _noticeShown = false;
   123→        if (!mounted) return;
   124→        ScaffoldMessenger.of(context)
   125→          ..hideCurrentSnackBar()
   126→          ..showSnackBar(SnackBar(content: Text(message)));
   127→      });
   128→    }
   129→  }

   149→  /// ينشئ عميلاً جديداً من المنتقي (P1-3) — بعملة السلة للعرض ورصيد
   150→  /// صفري (لا افتتاحي)؛ الفشل يُعاد للعرض داخل النموذج المصغّر.
   151→  Future<Result<CustomerPick, String>> _createQuickCustomer(
   152→    AppController app,
   153→    SellCartViewModel vm,
   154→    String name,
   155→    String? phone,
   156→  ) async {
   157→    final customers = app.customers;
   158→    final companies = app.companies;
   159→    if (customers == null || companies == null) {
   160→      return const Err<CustomerPick, String>(
   161→        'المستودعات غير جاهزة — أعد فتح الشاشة ثم حاول.',
   162→      );
   163→    }
   164→    final userId = (await companies.findAdminUserId()) ?? 1;
   165→    final result = await customers.createCustomer(
   166→      CustomerDraft(name: name, phone: phone),
   167→      userId: userId,
   168→    );
   169→    if (result.isErr) {
   170→      return Err<CustomerPick, String>(result.errorOrNull!);
   171→    }
   172→    return Ok<CustomerPick, String>(
   173→      CustomerPick(
   174→        partyId: result.valueOrNull!,
   175→        name: name,
   176→        phone: (phone ?? '').isEmpty ? null : phone,
   177→        currencyCode: vm.state.baseCurrency?.code ?? '',
   178→        balance: 0,
   179→        hasMoreCurrencies: false,
   180→      ),
   181→    );
   182→  }

   198→  Future<void> _openPayment() async {
   199→    final app = context.read<AppController>();
   200→    final vm = context.read<SellCartViewModel>();
   201→    final state = vm.state;
   202→    final priced = vm.pricedCart;
   203→    if (priced == null) return;
   204→    // P0-2: استهلاك invoicing.print_on_save (نمط بوابة الائتمان — قراءة
   205→    // لحظة فتح نافذة الدفع): ask = أزرار، always = معاينة تلقائية، off = إخفاء.
   206→    final printMode = await resolvePrintOnSave(app.settings);
   207→    if (!mounted) return;
   208→    final sales = app.sales;
   209→    final posted = await showPaymentSheet(
   210→      context,
   211→      grandTotal: priced.totals.grandTotal,
   212→      decimals: state.selectedCurrency?.decimals ?? 2,
   213→      currencyCode: state.selectedCurrency?.code,
   214→      customerName: state.customer?.name,
   215→      onConfirm: (paidCash, method) =>
   216→          vm.postSale(paidCash: paidCash, method: method),
   217→      // FR-03-05 (17-c): بوابة حد الائتمان — تُجلب لحظة التأكيد من
   218→      // المستودع (الحد + الرصيد الحي بعملة الفاتورة) والإعداد.
   219→      creditGate: () => _resolveCreditGate(app, vm),
   220→      // P0-2: فتح معاينة PDF للفاتورة المرحّلة (نمط زر PDF القائم).
   221→      printOnSave: printMode,
   222→      openInvoicePreview: sales == null
   223→          ? null
   224→          : (receipt) => openInvoicePdfPreview(
   225→              context,
   226→              saleRepo: sales,
   227→              invoiceId: receipt.invoiceId,
   228→            ),
   229→    );
   230→    if (posted) {
   231→      vm.dismissReceipt();
   232→    }
   233→  }

   235→  /// يجلب معطيات بوابة حد الائتمان — null عند غياب العميل/المستودعات
   236→  /// (عميل نقدي مجهول: لا آجل أصلاً فلا فحص — FR-03-05).
   237→  Future<CreditLimitGate?> _resolveCreditGate(
   238→    AppController app,
   239→    SellCartViewModel vm,
   240→  ) async {
   241→    final customerId = vm.state.customer?.id;
   242→    final currencyId = vm.state.currencyId;
   243→    if (customerId == null || currencyId == null) return null;
   244→    final customers = app.customers;
   245→    final settings = app.settings;
   246→    if (customers == null || settings == null) return null;
   247→    // checkCredit يقرأ الحد من جدول العميل والرصيد بصيغة FR-03-02
   248→    // بعملة الفاتورة (لا الحقل المخزَّن في السلة — دائماً حي).
   249→    final check = await customers.checkCredit(customerId, currencyId, 0);
   250→    final action = await settings.getString(
   251→      'parties.credit_limit_action',
   252→      'warn',
   253→    );
   254→    return CreditLimitGate(
   255→      creditLimit: check.creditLimit,
   256→      action: action,
   257→      currentBalance: check.balance,
   258→    );
   259→  }
