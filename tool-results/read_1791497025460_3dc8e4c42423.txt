   274→  Future<void> load() async {
   275→    if (_initialized) {
   276→      // إعادة الدخول للشاشة: تحديث سعر اليوم فقط (السلة كما هي).
   277→      await _refreshRateKnowledge();
   278→      return;
   279→    }
   280→    _state = SellCartState.initial;
   281→    notifyListeners();
   282→    try {
   283→      // سياسات التخصيص (UX-2a) — قراءة واحدة متزامنة مع باقي التحميل؛
   284→      // غياب المستودع يبقي سلوك v0.x (warn/خصومات ظاهرة/بلا تحذير هامش).
   285→      var overAvailPolicy = 'warn';
   286→      var showDiscounts = true;
   287→      var warnBelowMargin = false;
   288→      if (_settings != null) {
   289→        final policyResults = await Future.wait<Object?>([
   290→          _settings.overAvailPolicy(),
   291→          _settings.showDiscounts(),
   292→          _settings.discountBelowMargin(),
   293→        ]);
   294→        overAvailPolicy = policyResults[0]! as String;
   295→        showDiscounts = policyResults[1]! as bool;
   296→        warnBelowMargin = policyResults[2]! as bool;
   297→      }
   298→      final results = await Future.wait<Object?>([
   299→        _companies.listActiveCurrencies(),
   300→        _companies.findBaseCurrency(),
   301→        _companies.findDefaultWarehouseId(),
   302→        _companies.findAdminUserId(),
   303→      ]);
   304→      final currencies = results[0]! as List<Currency>;
   305→      final base = results[1] as Currency?;
   306→      final warehouseId = results[2] as int?;
   307→      _userId = results[3] as int?;
   308→
   309→      _initialized = true;
   310→      _state = SellCartState(
   311→        loading: false,
   312→        currencies: currencies,
   313→        baseCurrency: base,
   314→        currencyId: base?.id,
   315→        todayRate: null,
   316→        rateKnown: true,
   317→        customer: null,
   318→        warehouseId: warehouseId,
   319→        lines: const <CartUiLine>[],
   320→        invoiceDiscountType: SaleDiscountType.amount,
   321→        invoiceDiscountValue: 0,
   322→        posting: false,
   323→        nextInvoiceNo: await _nextInvoicePreview(),
   324→        overAvailPolicy: overAvailPolicy,
   325→        showDiscounts: showDiscounts,
   326→        warnBelowMargin: warnBelowMargin,
   327→      );

   330→        loading: false,
   331→        error: error,
   332→        currencies: const <Currency>[],
   333→        rateKnown: true,

   377→  /// أسطر السلة المباعة تحت التكلفة (UX-2a — لافتة تحذير بالكاشير عند
   378→  /// تفعيل `invoicing.discount_below_margin`): يقارن صافي السطر بعد كل
   379→  /// الخصومات بتكلفة الكمية — حكراً على عملة الأساس (التكلفة المحفوظة
   380→  /// بالأساس؛ فاتورة بعملة أخرى تُترك بلا تحذير كاذب).
   381→  List<String> get belowCostLineNames {
   382→    final state = _state;
   383→    if (!state.warnBelowMargin || state.lines.isEmpty) {
   384→      return const <String>[];
   385→    }
   386→    final pricedLines = pricedCart?.lines;
   387→    if (pricedLines == null || pricedLines.length != state.lines.length) {
   388→      return const <String>[];
   389→    }
   390→    final baseId = state.baseCurrency?.id;
   391→    if (baseId == null || state.currencyId != baseId) {
   392→      return const <String>[];
   393→    }
   394→    final names = <String>[];
   395→    for (var i = 0; i < state.lines.length; i++) {
   396→      final line = state.lines[i];
   397→      if (line.isService || line.costPrice <= 0) continue;
   398→      final costTotal = line.costPrice * line.qty;
   399→      if (pricedLines[i].netFinal < costTotal - _qtyEpsilon) {
   400→        names.add(line.name);
   401→      }
   402→    }
   403→    return names;
   404→  }

   435→  /// اختيار عملة الفاتورة — عند عملة غير الأساس بلا سعر اليوم تُفتح
   436→  /// بوابة الإدخال فوراً (FR-08-09).
   437→  Future<void> setCurrency(int currencyId) async {
   438→    if (currencyId == _state.currencyId) return;
   439→    final currency = _currencyById(currencyId);
   440→    final isBase = currency?.isBase ?? false;
   441→    var rateKnown = true;
   442→    double? rate;
   443→    if (!isBase) {
   444→      rate = await _fx.rateFor(currencyId, _clock());
   445→      rateKnown = rate != null && rate > 0;
   446→    }
   447→    _state = _stateWith(
   448→      currencyId: currencyId,
   449→      todayRate: rate,
   450→      rateKnown: rateKnown,
   451→      fxGateRequired: !isBase && !rateKnown,
   452→      postError: null,
   453→      notice: null,
   454→    );
   455→    notifyListeners();
   456→  }

   458→  /// حفظ سعر اليوم من بوابة FX (يكتب عبر fx.setRate ثم يتابع).
   459→  Future<Result<void, String>> saveTodayRate(double rate) async {
   460→    final currencyId = _state.currencyId;
   461→    if (currencyId == null) {
   462→      return const Err<void, String>('اختر عملة أولاً.');
   463→    }
   464→    final result = await _fx.setRate(
   465→      currencyId: currencyId,
   466→      date: _clock(),
   467→      rate: rate,
   468→      userId: _userId ?? 1,
   469→      now: _clock(),
   470→    );
   471→    if (result.isOk) {
   472→      await _refreshRateKnowledge();
   473→    }
   474→    return result;
   475→  }

   477→  /// استهلاك طلب فتح بوابة FX (بعد فتح النافذة).
   478→  void clearFxGate() {
   479→    if (!_state.fxGateRequired) return;
   480→    _state = _stateWith(fxGateRequired: false);
   481→    notifyListeners();
   482→  }
   483→
   484→  Future<void> _refreshRateKnowledge() async {
   485→    final currencyId = _state.currencyId;
   486→    if (currencyId == null) return;
   487→    final currency = _currencyById(currencyId);
   488→    if (currency == null || currency.isBase) {
   489→      _state = _stateWith(
   490→        todayRate: null,
   491→        rateKnown: true,
   492→        fxGateRequired: false,
   493→      );
   494→      notifyListeners();
   495→      return;
   496→    }
   497→    final rate = await _fx.rateFor(currencyId, _clock());
   498→    final known = rate != null && rate > 0;
   499→    _state = _stateWith(
   500→      todayRate: rate,
   501→      rateKnown: known,
   502→      fxGateRequired: known ? false : _state.fxGateRequired,
   503→    );
   504→    notifyListeners();
   505→  }
   506→
   507→  Currency? _currencyById(int id) {
   508→    for (final currency in _state.currencies) {
   509→      if (currency.id == id) return currency;
   510→    }
   511→    return null;
   512→  }

   518→  /// إضافة صنف (الكمية 1 أو دمج مع سطر موجود لنفس الصنف).
   519→  void addItemFromInfo(ItemStockInfo info) {
   520→    final warehouseId = _state.warehouseId;
   521→    final available = info.item.isService
   522→        ? null
   523→        : (info.warehouseQty[warehouseId] ?? info.totalQty);
   524→    final price = info.retailPrice ?? 0.0;
   525→    final existing = _lineIndexOf(info.item.id);
   526→    if (existing >= 0) {
   527→      _updateLine(existing, qty: _state.lines[existing].qty + 1);
   528→      return;
   529→    }
   530→    _state = _stateWith(
   531→      lines: [
   532→        ..._state.lines,
   533→        CartUiLine(
   534→          productId: info.item.id,
   535→          name: info.item.name,
   536→          barcode: info.item.barcode,
   537→          qty: 1,
   538→          unitPrice: price,
   539→          discountType: SaleDiscountType.amount,
   540→          discountValue: 0,
   541→          availableQty: available,
   542→          isService: info.item.isService,
   543→          // UX-2a: تكلفة اللقطة لكشف البيع تحت التكلفة (عملة الأساس حصراً).
   544→          costPrice: info.item.costPrice,
   545→        ),
   546→      ],
   547→      postError: null,
   548→    );

   550→  }

   625→  /// بناء مسودة الترحيل من الحالة الحالية (للاستخدام الداخلي والاختبارات).
   626→  SaleDraft? buildDraft({
   627→    required double paidCash,
   628→    required SalePaymentMethod method,
   629→  }) {
   630→    final currencyId = _state.currencyId;
   631→    final warehouseId = _state.warehouseId;
   632→    if (currencyId == null || warehouseId == null) return null;
   633→    return SaleDraft(
   634→      customerId: _state.customer?.id,
   635→      currencyId: currencyId,
   636→      lines: cartLines,
   637→      invoiceDiscountType: _state.invoiceDiscountType,
   638→      invoiceDiscountValue: _state.invoiceDiscountValue,
   639→      paidCash: paidCash,
   640→      paymentMethod: method,
   641→      warehouseId: warehouseId,
   642→      issuedAt: _clock(),
   643→    );
   644→  }

   646→  /// **ترحيل الفاتورة** — نجاح: تُفرَّغ السلة ويُحفظ الإيصال؛ فشل: تُعرض
   647→  /// رسالة الرفض العربية **والسلة لا تُفقد**.
   648→  Future<Result<SalePostedReceipt, String>> postSale({
   649→    required double paidCash,
   650→    required SalePaymentMethod method,
   651→  }) async {
   652→    if (_state.posting) {
   653→      return const Err<SalePostedReceipt, String>(
   654→        'ترحيل جارٍ بالفعل — لحظات ويكتمل.',
   655→      );
   656→    }
   657→    final failure = _prePostFailure(paidCash, method);
   658→    if (failure != null) {
   659→      _state = _stateWith(postError: failure);
   660→      notifyListeners();
   661→      return Err<SalePostedReceipt, String>(failure);
   662→    }
   663→    final draft = buildDraft(paidCash: paidCash, method: method);
   664→    if (draft == null) {
   665→      const message = 'الإعدادات غير مكتملة (العملة/المخزن) — أعد فتح الشاشة.';
   666→      _state = _stateWith(postError: message);
   667→      notifyListeners();
   668→      return const Err<SalePostedReceipt, String>(message);
   669→    }
   670→
   671→    _state = _stateWith(posting: true, postError: null);
   672→    notifyListeners();
   673→    final result = await _sales.postSale(
   674→      draft,
   675→      userId: _userId ?? 1,
   676→      now: _clock(),
   677→    );
   678→    if (result.isOk) {
   679→      final receipt = result.valueOrNull!;
   680→      _state = _stateWith(
   681→        posting: false,
   682→        lines: const <CartUiLine>[],
   683→        invoiceDiscountType: SaleDiscountType.amount,
   684→        invoiceDiscountValue: 0,
   685→        lastReceipt: receipt,
   686→        postError: null,
   687→        notice: null,
   688→        nextInvoiceNo: await _nextInvoicePreview(),
   689→      );
   690→      notifyListeners();
   691→    } else {
   692→      _state = _stateWith(posting: false, postError: result.errorOrNull!);
   693→      notifyListeners();
   694→    }
   695→    return result;
   696→  }

   698→  /// فحوص ما قبل الترحيل (بترتيب رسائل واضح للمستخدم) — null عند السلامة.
   699→  String? _prePostFailure(double paidCash, SalePaymentMethod method) {
   700→    if (_state.lines.isEmpty) {
   701→      return 'أضف بنداً واحداً على الأقل إلى الفاتورة قبل الدفع.';
   702→    }
   703→    final cartFailure = cartError;
   704→    if (cartFailure != null) return cartFailure;
   705→    final currency = _state.selectedCurrency;
   706→    if (currency == null) return 'اختر عملة الفاتورة أولاً.';
   707→    if (!currency.isBase && !_state.rateKnown) {
   708→      // FR-08-09: منع الحفظ + طلب فتح بوابة إدخال سعر اليوم.
   709→      _state = _stateWith(fxGateRequired: true);
   710→      notifyListeners();
   711→      return 'لا يوجد سعر صرف لعملة ${currency.code} بتاريخ اليوم — '
   712→          'أدخل سعر اليوم أولاً ثم أكمل الدفع.';
   713→    }
   714→    final total = grandTotal;
   715→    final payFailure = SalePricing.validatePayment(total, paidCash, method);
   716→    if (payFailure != null) return payFailure;
   717→    // UX-2a — `sale.over_avail_policy` = block: أي سطر يتجاوز المتاح يمنع
   718→    // الترحيل من هنا (warn = سلوك v0.x: تحذير بصري فقط والمستودع حارس
   719→    // أخير برسالته الخاصة).
   720→    if (_state.overAvailPolicy == 'block') {
   721→      for (final line in _state.lines) {
   722→        if (line.exceedsAvailable) {
   723→          return 'لا يمكن الترحيل: كمية «${line.name}» تتجاوز المتاح '
   724→              '(${_qtyText(line.availableQty!)}) — سياسة «منع البيع فوق '
   725→              'المتاح» مفعّلة من تفضيلات البيع. خفّض الكمية أو عوّض '
   726→              'المخزون أولاً.';
   727→        }
   728→      }
   729→    }
   730→    // عميل نقدي مجهول: نقدي كامل حصراً (الآجل يحتاج حساباً معروفاً).
   731→    final settlement = SalePricing.settlePayment(total, paidCash);
   732→    if (settlement.remainingCredit > moneyEpsilon &&
   733→        !_state.isCustomerSelected) {
   734→      return 'البيع الآجل يتطلب اختيار عميل أولاً — العميل النقدي المجهول '
   735→          'يسدد نقداً كاملاً فقط.';
   736→    }
   737→    return null;
   738→  }

   744→  /// مدة الصلاحية الافتراضية لعرض السعر المحفوظ من السلة (P2-5) —
   745→  /// +30 يوماً إن لم يمرّر المستدعي صلاحية صريحة.
   746→  static const Duration quotationDefaultValidity = Duration(days: 30);
   747→
   748→  /// حفظ السلة الحالية كعرض سعر QTE — بلا أي حركة مخزون/صندوق.
   749→  /// السلة تبقى كما هي (يمكن المتابعة في البيع أو التفريغ من الشاشة).
   750→  ///
   751→  /// [validUntil] (P2-5): صلاحية صريحة — والافتراضي +30 يوماً من لحظة
   752→  /// الحفظ (مسار الحفظ لا يمرّ بنموذج إدخال في هذه الجولة).
   753→  Future<Result<Quotation, String>> saveAsQuotation({
   754→    DateTime? validUntil,
   755→  }) async {
   756→    if (_state.lines.isEmpty) {
   757→      return const Err<Quotation, String>(
   758→        'أضف بنداً واحداً على الأقل قبل حفظ عرض السعر.',
   759→      );
   760→    }
   761→    final cartFailure = cartError;
   762→    if (cartFailure != null) {
   763→      return Err<Quotation, String>(cartFailure);
   764→    }
   765→    final currencyId = _state.currencyId;
   766→    final warehouseId = _state.warehouseId;
   767→    if (currencyId == null || warehouseId == null) {
   768→      return const Err<Quotation, String>(
   769→        'الإعدادات غير مكتملة (العملة/المخزن) — أعد فتح الشاشة.',
   770→      );
   771→    }
   772→    return _quotations.createQuotation(
   773→      QuotationDraft(
   774→        customerId: _state.customer?.id,
   775→        currencyId: currencyId,
   776→        warehouseId: warehouseId,
   777→        lines: [
   778→          for (final line in _state.lines)
   779→            QuotationLine(
   780→              productId: line.productId,
   781→              qty: line.qty,
   782→              unitPrice: line.unitPrice,
   783→              lineDiscountType: line.discountType,
   784→              lineDiscountValue: line.discountValue,
   785→            ),
   786→        ],
   787→        invoiceDiscountType: _state.invoiceDiscountType,
   788→        invoiceDiscountValue: _state.invoiceDiscountValue,
   789→        // P2-5: الصلاحية الافتراضية +30 يوماً عند غياب صلاحية صريحة.
   790→        validUntil: validUntil ?? _clock().add(quotationDefaultValidity),
   791→        issuedAt: _clock(),
   792→      ),
   793→      userId: _userId ?? 1,
   794→      now: _clock(),
   795→    );
   796→  }
